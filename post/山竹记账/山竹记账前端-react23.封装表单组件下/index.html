import type { JSONValue } from '@/global'
import { create } from 'zustand'
import { formValidate, type FormErrorsType, type FormRules } from '@/lib/validator/formValidate'

export type FormStore<TData, TForm extends JSONValue> = {
  data: TData
  error: FormErrorsType<TForm>
  setData: (data: Partial<TData>) => void
  setError: (error: Partial<FormErrorsType<TForm>>) => void
  clearError: (key?: keyof TForm) => void
  revalidateFormItems: <T extends TForm>(
    formData: Partial<T>,
    formRules: FormRules<T>,
    lastError?: FormErrorsType<T>
  ) => void
}

export function createFormStore<TData, TForm extends JSONValue>(
  options: {
    genInitData: () => TData
    genInitError: () => FormErrorsType<TForm>
  }
) {
  return create<FormStore<TData, TForm>>()((set, get) => ({
    data: options.genInitData(),
    error: options.genInitError(),
    setData: (data: Partial<TData>) => {
      set((state) => ({
        ...state,
        data: { ...state.data, ...data },
      }))
    },
    setError: (error: Partial<FormErrorsType<TForm>>) => {
      set((state) => ({
        ...state,
        error: { ...state.error, ...error },
      }))
    },
    clearError: (key?: keyof TForm) => {
      set((state) => ({
        ...state,
        error: key ? { ...state.error, [key]: [] } : options.genInitError(),
      }))
    },
    revalidateFormItems: <T extends TForm>(
      formData: Partial<T>,
      formRules: FormRules<T>,
      lastError?: FormErrorsType<T>
    ) => {
      get().clearError()
      const keys = Object.keys(formData ?? {})
      const validatingRules = formRules.filter((rule) => keys.includes(String(rule.key)))
      const formError = formValidate(formData, validatingRules)
      get().setError({ ...lastError, ...formError } as Partial<FormErrorsType<TForm>>)
    },
  }))
}

import { useTagCreateStore } from '@/stores/useTagCreateStore'
import { useTagEditStore } from '@/stores/useTagEditStore'

interface Props {
  className?: string
  mode: 'create' | 'edit'
  submitButtonText?: string
}

const tagRules: FormRules<PickedTag> = [
  { key: 'name', type: 'required', required: true, message: '标签名称不能为空' },
  { key: 'name', type: 'length', min: 1, max: 4, message: '最多输入4个字符' },
  { key: 'sign', type: 'required', required: true, message: '符号不能为空' },
] as const

export const TagForm: React.FC<Props> = ({
  className,
  mode = 'create',
  submitButtonText = '确定',
}) => {
  const create = useTagCreateStore()
  const edit = useTagEditStore()
  const {
    data: lastData,
    error: lastError,
    setData,
    clearError,
    revalidateFormItems,
  } = mode === 'edit' ? edit : create

  const onChangeTagName = (name: string) => {
    setData({
      ...lastData,
      name,
    })
  }
  const onBlurTagName = () => {
    revalidateFormItems<PickedTag>({ name: lastData.name }, tagRules, lastError)
  }
  const onFocusTagName = () => {
    clearError('name')
  }

  const onChangeTagSign = (sign: string) => {
    setData({
      sign,
    })
  }

  const onBlurTagSign = () => {
    revalidateFormItems<PickedTag>({ sign: lastData.sign }, tagRules, lastError)
  }
  const onFocusTagSign = () => {
    clearError('sign')
  }

  const onSubmit: FormEventHandler<HTMLFormElement> = async (e) => {
    e.preventDefault()
    revalidateFormItems<PickedTag>(lastData, tagRules, lastError)
    // 注意过期的闭包 lastData、lastError 都是闭包，不能直接使用，需要用 useSignInStore.getState() 获取最新状态
    const activeStore = mode === 'edit' ? useTagEditStore : useTagCreateStore
    const freshError = activeStore.getState().error
    const freshData = activeStore.getState().data
    if (!hasError(freshError)) {
      console.log('onSubmit_______________________')
      console.log('%c⧭ lastData', 'color: #40fff2', lastData)
      console.log('%c⧭ freshData', 'color: #40fff2', freshData)
      console.log('%c⧭ lastError', 'color: #40fff2', lastError)
      console.log('%c⧭ freshError', 'color: #40fff2', freshError)
      // console.log('%c⧭ TODO: 发送创建标签请求', 'color: #40fff2', formData)
      console.log('_______________________onSubmit')
    }
  }

  return (
    <form onSubmit={onSubmit} className={cn('form-sc', className)}>
      <FormItem
        formItemType="input"
        formItemTypeProps={{
          inputType: 'text',
          label: '标签名',
          labelFor: 'tag-name',
          placeholder: '请输入标签名称',
          ifUseLabelTag: true,
          errorHint: lastError?.name && lastError.name.join('，'),
          value: lastData.name,
          onChange: onChangeTagName,
          onBlur: onBlurTagName,
          onFocus: onFocusTagName,
        }}
      />

      <FormItem
        formItemType="emoji"
        formItemTypeProps={{
          errorHint: lastError?.sign && lastError.sign.join('，'),
          value: lastData.sign,
          onChange: onChangeTagSign,
          onBlur: onBlurTagSign,
          onFocus: onFocusTagSign,
        }}
      />

      <p className="form-item-sc text-center">记账时长按，即可进行编辑</p>

      <div className="form-item-sc">
        <button type="submit" className="primary-btn">
          {submitButtonText}
        </button>
      </div>

      {mode === 'edit' && (
        <div className="form-item-sc mb-32px">
          <button type="submit" className="primary-btn bg-[var(--delete-button-bg)]">
            删除标签
          </button>
        </div>
      )}
    </form>
  )
}
  • useTagCreateStore、useTagEditStore 必须以静态形式提前声明，不可放到条件判断中
    • ❌ const store = mode === 'edit' ? useTagEditStore() : useTagCreateStore() 会报钩子使用错误
    • ✅ const create = useTagCreateStore(); const edit = useTagEditStore(); cosnt store = mode === 'edit' ? edit : create;

import { emojiTopicList, type EmojiMainTopic } from '@/lib/const/emojiTopicList'
import { safeEmojiList, type EmojiTopic } from '@/lib/const/safeEmojiList'

type EmojiItem = {
  id: number
  emoji: string
  hint: string
  topic: EmojiTopic['topic']
  mainTopic: EmojiMainTopic
}

export const getEmojiById = (id: number): EmojiItem => {
  const emojiItem = safeEmojiList.find((item) => {
    return item.emoji_list.find((emoji) => emoji.index === id)
  })

  if (!emojiItem) {
    throw new Error('emoji not found by getEmojiById')
  }

  const emoji = emojiItem?.emoji_list.find((emoji) => emoji.index === id)?.unicode ?? ''
  const hint = emojiItem?.emoji_list.find((emoji) => emoji.index === id)?.hint ?? ''
  const topic = emojiItem?.topic
  const mainTopic = emojiTopicList.find((item) =>
    (item.topics as readonly string[]).includes(topic)
  )?.mainTopic

  if (!mainTopic) {
    throw new Error('mainTopic not found by getEmojiById')
  }

  return {
    id,
    emoji,
    hint,
    topic,
    mainTopic,
  }
}
  • 根据 safeEmojiList 中各个 emoji 的 index 属性返回选中的 emojiItem 信息
  • 可以直接将这个数据属性渲染到页面上，避免各层次的组件传参

import type { EmojiMainTopic } from '@/lib/const/emojiTopicList'
import type { EmojiTopic } from '@/lib/const/safeEmojiList'
import { create } from 'zustand'
import { withDevtoolsBridge } from 'zustand-devtools-bridge'

interface EmojiState {
  id?: number | null
  emoji?: string
  hint?: string
  topic?: EmojiTopic['topic'] | ''
  mainTopic?: EmojiMainTopic
}

interface EmojiActions {
  setEmoji: (emojiState: EmojiState) => void
  reset: () => void
}

const initialEmojiState: EmojiState = {
  id: null,
  emoji: '',
  hint: '',
  topic: '',
  mainTopic: '表情',
}

export const useEmojiStore = create<EmojiState & EmojiActions>()(
  withDevtoolsBridge(
    (set, _get) => ({
      ...initialEmojiState,
      setEmoji: (emojiState: Partial<EmojiState>) => set((state) => ({ ...state, ...emojiState })),
      reset: () => set(initialEmojiState),
    }),
    {
      name: 'emoji',
      enabled: import.meta.env.DEV,
    }
  )
)
  • 使用 zustand-devtools-bridge 浏览器插件配合插件库方便在控制台调试，实时查看状态数据
  • 主要需要的数据 id: emoji_id、emoji、hint、topic、mainTopic
  • 模拟从请开启中获取到 sign_id 后，就可以通过 getEmojiById 获取 emojiItem 数据
  • 再更新到此状态中，以供 emoji 各个相关组件使用

export type FormItemType = 'input' | 'emoji' | 'validateCodeSend'
export type InputType = 'text' | 'password' | 'email' | 'number'
export type CommonFormItemTypeProps = {
  label?: string
  labelFor?: string
  ifUseLabelTag?: boolean
  hasLabelColon?: boolean
  placeholder?: string
  errorHint?: string
  hasErrorPlaceholder?: boolean
  value?: string
  onChange?: (value: string) => void
  onBlur?: () => void
  onFocus?: () => void
}

export type FormItemTypeProps<T extends FormItemType> = T extends 'input'
  ? CommonFormItemTypeProps & { type?: 'input'; inputType: InputType }
  : T extends 'emoji'
    ? Omit<CommonFormItemTypeProps, 'value' | 'onChange'> & {
        type?: 'emoji'
        value?: number
        onChange?: (sign_in: number, sign: string) => void
      }
    : T extends 'validateCodeSend'
      ? CommonFormItemTypeProps & { type?: 'validateCodeSend'; onClickSendCode?: () => void }
      : never
  • 注意合并类型别名，避免合并同名属性为 never，需要合并覆盖时使用 Omit 忽略同名属性

interface Props<T extends FormItemType> {
  className?: string
  formItemType: T
  formItemTypeProps: FormItemTypeProps<T>
}

export const FormItem = <T extends FormItemType>({
  className,
  formItemType,
  formItemTypeProps,
}: Props<T>) => {
  const renderContent = () => {
    switch (formItemType) {
      case 'input':
        return (
          <InputText {...(formItemTypeProps as FormItemTypeProps<'input'>)} className={className} />
        )
      case 'emoji':
        return (
          <InputEmoji
            {...(formItemTypeProps as FormItemTypeProps<'emoji'>)}
            className={className}
          />
        )
      case 'validateCodeSend':
        return (
          <InputValidateCodeSend
            {...(formItemTypeProps as FormItemTypeProps<'validateCodeSend'>)}
            className={className}
          />
        )
      default:
        return null
    }
  }
  return <>{renderContent()}</>
}
  • TS 在 switch (formItemType) 中不会把泛型 T 缩窄为字面量

interface Props {
  className?: string
  hasLabelColon?: boolean
  errorHint?: string
  hasErrorPlaceholder?: boolean
  value?: number
  onChange?: (sign_id: number, sign: string) => void
  onBlur?: () => void
  onFocus?: () => void
}

export const InputEmoji: React.FC<Props> = ({
  className,
  hasLabelColon = true,
  errorHint,
  hasErrorPlaceholder = true,
  value = '',
  onChange,
  ...restProps
}) => {
  const { id: emojiId, emoji, hint, mainTopic, setEmoji } = useEmojiStore()

  useEffect(() => {
    if (!emojiId || !emoji) {
      return
    }
    const emojiItem = getEmojiById(emojiId)
    setEmoji(emojiItem)
    onChange?.(emojiId ?? 0, emoji ?? '')
  }, [emojiId, value, emoji, mainTopic, onChange, setEmoji])

  return (
    <section className={cn('form-item-sc relative', className)}>
      <div className="form-label-sc text-[22px]">
        <span className="text-[16px]">符号 {hasLabelColon ? '：' : ''}</span>
        <span className={cn('text-22px inline-block wag')}>{emoji}</span>
        <span className="text-[16px] animate-rainbow color-[var(--vp-c-brand-next)]">{hint}</span>
      </div>

      <EmojiSelector {...restProps} hasError={!!errorHint} className="form-item-content-sc" />

      {(errorHint || hasErrorPlaceholder) && (
        <span className="form-item-error-sc w-fit buzz">{errorHint ?? <>&nbsp;</>}</span>
      )}
    </section>
  )
}
  • 监听状态数据 id: emojiId、 emoji、 mainTopic，和外部数据传参 value
    • 提前退出逻辑 !emojiId || !emoji
    • 通过 id 获取 emojiItem 数据，更新状态
    • 并且使用外部传参回调函数 onChange 传参数据 emojiId、emoji，更新表单数据
  • 通过 {...restProps} 透传事件 onBlur、onFocus 到子组件 EmojiSelector
  • 不与子组件直接通信，而是和状态 useEmojiStore 通信
