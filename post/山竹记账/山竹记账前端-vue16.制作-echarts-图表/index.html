import {
  type ComputedRef,
  type PropType,
  defineComponent,
  reactive,
  computed,
  ref,
} from 'vue'
import { MainLayout } from '@/components/layout/MainLayout.tsx'
import { SwitchTabs } from '@/components/tab/SwitchTabs.tsx'
import { ItemTab } from '@/components/tab/ItemTab.tsx'
import { CustomDateRangeOverlay } from '@/modules/items/CustomDateRangeOverlay.tsx'
import { DateTime } from '@/shared/dateTime.ts'

interface DateTab {
  tabName: string
  tabId: 'thisMonth' | 'lastMonth' | 'thisYear' | 'customDateRange'
  start: string
  end: string
}

export const DateTabsLayout = defineComponent({
  name: 'DateTabsLayout',
  props: {
    title: { type: String as PropType<string>, default: '', },
  },
  setup( _props, _ctx ) {
    const currentDate = new DateTime()
    const customDate = reactive({
      start: currentDate.getOriginDate()?.format() || '',
      end: currentDate.getOriginDate()?.format() || '',
    })
    const itemTabList: ComputedRef<DateTab[]> = computed(() => [
      {
        tabName: '本月',
        tabId: 'thisMonth',
        start: currentDate.firstDayOfMonth().format(),
        end: currentDate.lastDayOfMonth().format(),
      },
      {
        tabName: '上月',
        tabId: 'lastMonth',
        start: currentDate.subtract(1, 'month').firstDayOfMonth().format(),
        end: currentDate.subtract(1, 'month').lastDayOfMonth().format(),
      },
      {
        tabName: '今年',
        tabId: 'thisYear',
        start: currentDate.firstDayOfYear().format(),
        end: currentDate.lastDayOfYear().format(),
      },
      {
        tabName: '自定义日期范围',
        tabId: 'customDateRange',
        start: customDate.start,
        end: customDate.end,
      },
    ])
    const selectedTabIdRef = ref<DateTab['tabId']>('thisMonth')
    const overlayVisibleRef = ref(false)
    const openCustomDateRange = () => {
      overlayVisibleRef.value = selectedTabIdRef.value === 'customDateRange'
    }
    return () => (
      <MainLayout>
        {{
          title: () => `${_props.title}`,
          icon: () => _ctx.slots?.icon?.(),
          default: () => (
            <div class="h-full flex flex-col">
              <SwitchTabs
                v-model:selectedTabId={selectedTabIdRef.value}
                onUpdate:selectedTabId={openCustomDateRange}
                class="[&>nav>menu>li]:(text-start grow-0)">
                {
                  itemTabList.value
                    .map(tabItem => (
                      <ItemTab key={tabItem.tabId} tabName={tabItem.tabName} tabId={tabItem.tabId}>
                        {_ctx.slots?.default?.({ start: tabItem.start, end: tabItem.end })}
                      </ItemTab>
                    ))
                }
              </SwitchTabs>

              {/* 自定义日期范围弹出层 */}
              <CustomDateRangeOverlay
                v-model:customStartDate={customDate.start}
                v-model:customEndDate={customDate.end}
                v-model:visibleValue={overlayVisibleRef.value} />
            </div>
          ),
        }}
      </MainLayout>
    )
  },
})
  • 保证渲染动态节点的数据必须是响应式的，比如 ref、reactive、computed 即代码中遍历渲染的 itemTabList 数据
  • v-model 绑定的数据也必须是响应式的，比如 ref、reactive的整体或其中某个属性、可读写的 computed 值

import { computed, defineComponent, type PropType, ref } from 'vue'
import { Overlay } from 'vant'
import { FormContainer } from '@/components/form/FormContainer.tsx'
import { FormItem } from '@/components/form/FormItem.tsx'
import cssVars from '@/assets/styles/vars.module.scss'

export const CustomDateRangeOverlay = defineComponent({
  name: 'CustomDateRangeOverlay',
  props: {
    visibleValue: { type: Boolean as PropType<boolean>, default: false, },
    customStartDate: { type: String as PropType<string>, default: '', },
    customEndDate: { type: String as PropType<string>, default: '', },
  },
  emits: ['update:visibleValue', 'update:customStartDate', 'update:customEndDate',],
  setup( _props, _ctx ) {
    const {zIndexFloatOverlay,} = cssVars
    const getVisibleValue = computed({
      get() {return _props.visibleValue},
      set(value) {_ctx.emit('update:visibleValue', value)}
    })
    const lastStarDate = ref(_props.customStartDate)
    const lastEndDate = ref(_props.customEndDate)
    const getCustomStartDate = computed({
      get() {return _props.customStartDate},
      set(value) {_ctx.emit('update:customStartDate', value)}
    })
    const getCustomEndDate = computed({
      get() {return _props.customEndDate},
      set(value) {_ctx.emit('update:customEndDate', value)}
    })
    const onSubmitCustomDateRange = (e: SubmitEvent) => {
        e.preventDefault()
        _ctx.emit('update:visibleValue', false)
        _ctx.emit('update:customStartDate', getCustomStartDate.value)
        _ctx.emit('update:customEndDate', getCustomEndDate.value)
        lastStarDate.value = getCustomStartDate.value
        lastEndDate.value = getCustomEndDate.value
      }
    const onCancelCustomDateRange = () => {
      _ctx.emit('update:visibleValue', false)
      getCustomStartDate.value = lastStarDate.value
      getCustomEndDate.value = lastEndDate.value
    }
    const startDateVisibleRef = ref(false)
    const endDateVisibleRef = ref(false)
    return () => (
      <Overlay
        z-index={zIndexFloatOverlay}
        show={getVisibleValue.value}
        onClick={onCancelCustomDateRange}
        class="flex items-center justify-center bg-[var(--overlay-mask-bg)]" >
        <div
          onClick={(e: MouseEvent) => e.stopPropagation()}
          class="bg-[var(--overlay-bg)] w-80vw h-[clamp(340px,35vh,400px)]">
          <header class="bg-[var(--main)] text-[var(--dialog-header-text)] py-12px px-1em">
            请选择时间
          </header>
          <main>
            <FormContainer onSubmit={onSubmitCustomDateRange}>
              <FormItem
                type="date"
                label="开始时间"
                v-model={getCustomStartDate.value}
                v-model:visibleValue={startDateVisibleRef.value} />
              <FormItem
                type="date"
                label="结束时间"
                v-model={getCustomEndDate.value}
                v-model:visibleValue={endDateVisibleRef.value} />
              <footer class="flex justify-between mt-1em children:(w-[calc(50%-.25em)])">
                <FormItem
                  type="button"
                  onClick={onCancelCustomDateRange}
                  buttonConfig={{level: 'default', text: '取 消' }}/>
                <FormItem
                  type="submit"
                  buttonConfig={{level: 'primary', text: '确 认' }}/>
              </footer>
            </FormContainer>
          </main>
        </div>
      </Overlay>
    )
  },
})
  • 取消时恢复上一次记录的值，就必须另外声明响应式变量 lastStarDate 和 lastEndDate
  • 限制弹出框高度大小适应范围 h-[clamp(340px,35vh,400px)] 的样式
  • 暂时缺少日期验证逻辑：结束时间必须大于开始时间

import { defineComponent, type PropType, } from 'vue'
import { ItemTab } from '@/components/tab/ItemTab.tsx'
import { cn } from 'cn'
import { genDefaultSlotTabs } from '@/components/tab/genDefaultSlotTabs.ts'

export const SwitchTabs = defineComponent({
  name: 'SwitchTabs',
  props: {
    selectedTabId: {
      type: String as PropType<string>,
    },
  },
  emits: [
    // 由点击事件触发
    'update:selectedTabId',
  ],
  setup(_props, _ctx) {
    const selectedClass = (tabId: string) => _props.selectedTabId === tabId
      ? cn(
          'relative',
          `after:(content-[""] block absolute bottom-0 left-0 w-full h-4px bg-[var(--tabs-indicator-bg)])`,
        )
      : ''
    const onSwitchTab = (tabId: string) => {
      _ctx.emit('update:selectedTabId', tabId)
    }

    return () => {
      // 支持插槽使用map遍历生成的文档片段
      const defaultSlotTabs =
        genDefaultSlotTabs(_ctx.slots?.default?.() ?? [])
      const ifAllItemTab = defaultSlotTabs
        .every((slot) =>
          slot.type === ItemTab &&
          slot.props &&
          Object.hasOwn(slot.props,'tabId') &&
          Object.hasOwn(slot.props,'tabName')
        )
      if (!ifAllItemTab) {
        throw new Error(`SwitchTabs: 子组件必须是 ItemTab 组件，且必须有 tabId 和 tabName 属性`)
      }

      return (
        <div class="grow-1 shrink-1 overflow-auto scrollbar-hide relative">
          {/* 切换标签导航 */}
          <nav class="sticky left-0 top-0 z-1 bg-[var(--navbar-bg-end)]">
            <menu class="flex justify-start items-center text-color-[var(--navbar-text-color)]">
              {
                defaultSlotTabs
                  .map((slot, index) => (
                    <li
                      key={`${index}-${slot?.props?.tabId}`}
                      class={cn(
                        'where:(text-center grow-1 shrink-0 px-1em py-12px)',
                        selectedClass(slot?.props?.tabId),
                      )}
                      onClick={() => onSwitchTab(slot?.props?.tabId)}>
                      {slot?.props?.tabName}
                    </li>
                  ))
              }
            </menu>
          </nav>

          {/* 标签内容-插槽-数组 */}
          <section class="max-h-[calc(100vh-136px)] overflow-auto scrollbar-hide">
            {
              defaultSlotTabs
                .find((slot) =>
                  slot.props?.tabId === _props.selectedTabId)
            }
          </section>
        </div>
      )
    }
  },
})
  • 将标签选择机制从 tabName 改为 tabId
  • 将 selected 的 prop 重命名为 selectedTabId
  • 将 update:selected 事件改为 update:selectedTabId
  • 内部比较逻辑统一使用 tabId 进行匹配，tabName 仅用来展示tab文本
  • 其他逻辑不变
    • 内部必须使用默认插槽接收 ItemTab 组件，ItemTab 组件必须有 tabId 和 tabName 属性
    • 拿到默认插槽的子组件数组，遍历判断是否为 ItemTab 或是 v-fgt，递归拍平返回子组件数组并渲染 defaultSlotTabs
  • 需要重构引用它的相关组件 @src/views/ItemCreate.tsx

import { defineComponent, } from 'vue'
import { ChartsSummary } from '@/modules/statistics/ChartsSummary.tsx'
import { DateTabsLayout } from '@/components/layout/DateTabsLayout.tsx'
import { SvgIcon } from '@/components/icons/SvgIcon.tsx'

export const StatisticsView = defineComponent({
  name: 'StatisticsView',
  setup(/* _props, _ctx */) {
    return () => (
      <DateTabsLayout title="统计图表">
        {{
          icon: () => <SvgIcon name="back-arrow" class="size-1.5em" />,
          default: ({ start, end }: { start: string; end: string }) => (
            <ChartsSummary startDate={start} endDate={end} />
          ),
        }}
      </DateTabsLayout>
    )
  },
})
  • 如果需要将一个组件传参到子组件内部的话，直接使用插槽，代替通过使用外部参数 props 来传一个组件，后者需要声明组件类型
    • 还可以定义好插槽执行函数的参数，即作用域插槽
    • default: ({ start, end }: { start: string; end: string }) => (<Xxx startDate={start} endDate={end} />)
  • 子组件消费插槽时，传递实参 {_ctx.slots?.default?.({ start: tabItem.start, end: tabItem.end })}
