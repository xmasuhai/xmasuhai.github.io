import { SvgIcon } from "@/components/SvgIcon";
import { useNumpadButtons } from "@/modules/items/input/useNumpadButtons.tsx";
import { cn } from "cn";
import s from './Numpad.module.scss'

interface Props {
  className?: string
}

export const Numpad: React.FC<Props> = ({ /* className, */ }) => {
  const numberSizeClass = (text: string) =>
  (/[0-9|dot]/g.test(text)
    ? cn('text-18px')
    : '')
  const { numpadButtons } = useNumpadButtons()

  return (
    <div
      className={cn(
        'grid grid-cols-[repeat(4,1fr)] grid-rows-[repeat(4,56px)]',
        'where:(bg-[var(--pad-border-color)]) gap-1px p-t-1px',
      )}
      style={{
        'gridTemplateAreas': `
          'btn_1 btn_2 btn_3 btn_clear'
          'btn_4 btn_5 btn_6 btn_backspace'
          'btn_7 btn_8 btn_9 btn_submit'
          'btn_0 btn_0 btn_dot btn_submit'
          `,
      }}>
      {numpadButtons.map((buttonItem) => (
        <button
          type="button"
          key={buttonItem.text}
          data-btn-text={`btn_${buttonItem.id}`}
          onClick={buttonItem.onClick}
          style={{
            'gridArea': `btn_${buttonItem.id}`,
          }}
          className={cn(
            'flex justify-center items-center font-bold',
            'bg-[var(--pad-button-bg-color)] b-unset',
            buttonItem?.extraClass,
            numberSizeClass(buttonItem.text),
            s['button-item'],
          )}>
          {buttonItem.hasIcon ? <SvgIcon name={buttonItem.iconName} /> : buttonItem.text}
        </button>
      ))}
    </div>
  )
}
  • 注意 React.JSX，正确使用属性名称，否则会报错 'grid-area'”不在类型“Properties<string | number, string & {}>
    • 使用 className，而不是 class
    • 使用 gridTemplateAreas，而不是 grid-template-areas
    • 使用 gridArea，而不是 grid-area
  • 使用网格布局 grid grid-cols-[repeat(4,1fr)] grid-rows-[repeat(4,56px)]
    • 使用具体的高度数值撑开按钮高度 grid-rows-[repeat(4,56px)]；宽度平均分配即可
    • 在容器中使用 gridTemplateAreas 指定具名按钮位置
    • 在每一个网格，即按钮上，使用 gridArea 指定按钮名称；
    • 遍历 numpadButtons 指定 gridArea 为 btn_${buttonItem.id}
    • 和 gridTemplateAreas 的名称位置一一对应
    • 比起一个个按钮元素上指定 row-start-* col-start-* row-end-* col-end-* 更直观
    • 缺点是无法使用 CSS Grid Generator Tool 直接获取
  • 键盘整体颜色设置为 #ddd，并且设置间隔 gap-1px，各个按钮背景色为白色，从而透出类似边框的样式，这样不用考虑边框重合问题
  • 使用按钮数据 const { numpadButtons } = useNumpadButtons()
  • 可使用图标/文字作为按钮内容 {buttonItem.hasIcon ? <SvgIcon name={buttonItem.iconName} /> : buttonItem.text}

import { OverlayMask } from "@/components/menu/OverlayMask"
import { animated, useSpring } from "@react-spring/web"
import { cn } from "cn"

interface Props {
  className?: string
  visible: boolean
  onClickMask?: () => void
}

export const PopUp: React.FC<Props> = ({ className, visible, onClickMask, }) => {
  const menuStyle = useSpring({
    transform: visible ? 'translateY(0%)' : 'translateY(100%)',
    opacity: visible ? 1 : 0,
    config: {
      tension: 200,
      friction: 20,
      duration: 180,
    },
  })

  return (
    <div className={cn(
      'z-[var(--z-popup)]',
      className,
    )}>
      {/* 遮罩层 */}
      <OverlayMask
        visible={visible}
        onClickMask={onClickMask}
        className="bg-black:50" />

      {/* 内容 */}
      <animated.div
        style={menuStyle}
        className={cn(
          'fixed bottom-0 left-0 w-full min-h-300px bg-red',
          'z-[calc(var(--z-popup)+1)]',
        )}>
        visible {`${visible}`}
      </animated.div>
    </div>
  )
}
  • 可复用 OverlayMask 组件
  • useSpring 动画配置从下往上显示 transform: visible ? 'translateY(0%)' : 'translateY(100%)',
  • 调整 z-index 层级 z-[var(--z-popup)]

export const usePopup = () => {
  const [visible, setVisible] = useState(false)

  const showPopup = () => { setVisible(true) }

export const usePopup = () => {
  const [visible, setVisible] = useState(false)

  const showPopup = () => { setVisible(true) }
  const hidePopup = () => { setVisible(false) }
  const togglePopup = () => { setVisible(!visible) }
  const popup = <PopUp visible={visible} onClickMask={() => hidePopup()} />
  const portalPopup = createPortal(popup, document.body)
  return {
    popup,
    portalPopup,
    showPopup,
    hidePopup,
    togglePopup,
  }
}

type Parts = {
  year: number
  month: number
  day: number
  hour: number
  minute: number
  second: number
  ms: number
}
type Unit = 'year' | 'month' | 'day' | 'hour' | 'minute' | 'second' | 'ms'
type DateTimeParams = number | string | Date

export class DateTime {
  #date: Date
  constructor(p?: DateTimeParams) {
    this.#date = p ? new Date(p) : new Date()
  }
  // t = new DateTime(); t.parts.year;
  get parts(): Parts {
    return {
      year: this.#date.getFullYear(),
      month: this.#date.getMonth() + 1,
      day: this.#date.getDate(),
      hour: this.#date.getHours(),
      minute: this.#date.getMinutes(),
      second: this.#date.getSeconds(),
      ms: this.#date.getMilliseconds(),
    }
  }

  // p = {year: 2000, month: 10}; t.parts = p;
  set parts(p: Partial<DateTime['parts']>) {
    const table = {
      year: 'setFullYear',
      month: 'setMonth',
      day: 'setDate',
      hour: 'setHours',
      minute: 'setMinutes',
      second: 'setSeconds',
      ms: 'setMilliseconds',
    } as const
    Object.entries(p).forEach(([key, value]) => {
      const methodsName = table[key as keyof typeof table]
      value = key === 'month' ? value - 1 : value
      this.#date[methodsName](value)
    })
  }

  get year(): number {
    return this.parts.year
  }
  set year(value: number) {
    this.parts = { year: value }
  }
  get month(): number {
    return this.parts.month
  }
  set month(value: number) {
    this.parts = { month: value }
  }
  get day(): number {
    return this.parts.day
  }
  set day(value: number) {
    this.parts = { day: value }
  }
  get hour(): number {
    return this.parts.hour
  }
  set hour(value: number) {
    this.parts = { hour: value }
  }
  get minute(): number {
    return this.parts.minute
  }
  set minute(value: number) {
    this.parts = { minute: value }
  }
  get second(): number {
    return this.parts.second
  }
  set second(value: number) {
    this.parts = { second: value }
  }
  get ms(): number {
    return this.parts.ms
  }
  set ms(value: number) {
    this.parts = { ms: value }
  }

  // t.add(-10, 'year').year
  add(n: number, unit: Unit) {
    const table = {
      year: 'year',
      month: 'month',
      day: 'day',
      hour: 'hour',
      minute: 'minute',
      second: 'second',
      ms: 'ms',
    } as const
    this[table[unit]] += n
    return this
  }
  /**
   * 格式化输出
   * https://tc39.es/ecma262/multipage/numbers-and-dates.html#sec-date-time-string-format
   * @pattern 目前只支持 YYYY-MM-DDTHH:mm:ss.sssZ
   * @memberof DateTime
   * @example (new DateTime()).format('YYYY-MM-DDTHH:mm:ss.sssZ') DateTime
   * @example (new DateTime()).format('YYYY年MM月DD日') DateTime
   */
  format(pattern = 'YYYY-MM-DD') {
    return pattern
      .replace(/YYYY/g, this.year.toString())
      .replace(/MM/g, this.month.toString().padStart(2, '0'))
      .replace(/DD/g, this.day.toString().padStart(2, '0'))
      .replace(/HH/g, this.hour.toString().padStart(2, '0'))
      .replace(/mm/g, this.minute.toString().padStart(2, '0'))
      .replace(/ss/g, this.second.toString().padStart(2, '0'))
      .replace(/sss/g, this.ms.toString().padStart(3, '0'))
  }
}

    setTranslateY(translateY + deltaY)
    setLastY(currentY)

          'absolute top-[calc(50%-var(--half-date-item-height))] left-0',

import { animated, useSpring } from '@react-spring/web'

interface Props {/* ... */}

export const DatePickerPopup: React.FC<Props> = ({
  className,
  start,
  end,
  value,
  dateItemHeight = 36,
}) => {
  const startTime = start ? createDateTime(start) : createDateTime().add(-3, 'year')
  const endTime = end ? createDateTime(end) : createDateTime().add(3, 'year')
  if (startTime.getTimestamp() >= endTime.getTimestamp()) {
    throw new Error('startTime must be greater than endTime')
  }

  const valueTime = value ? createDateTime(value) : createDateTime()
  const yearList = Array.from({ length: endTime.year - startTime.year + 1 }).map(
    (_, i) => startTime.year + i
  )
  const halfDateItemHeight = dateItemHeight / 2
  const itemHeightVarStyle = {
    '--date-item-height': `${dateItemHeight}px`,
    '--half-date-item-height': `${halfDateItemHeight}px`,
    '--total-height': `${yearList.length * dateItemHeight}px`,
  }
  const currentYearIndex = yearList.indexOf(valueTime.year)

  const [isTouching, setIsTouching] = useState(false)
  const [lastY, setLastY] = useState(-1)
  const [translateY, setTranslateYState] = useState(currentYearIndex * -dateItemHeight)

  const spring = useSpring({
    transform: `translateY(${translateY}px)`,
    immediate: isTouching,
    config: {
      tension: 300,
      friction: 30,
    },
  })

  const setTranslateY = (y: number) => {
    y = Math.min(0, y)
    y = Math.max(y, (yearList.length - 1) * -dateItemHeight)
    setTranslateYState(y)
  }

  const onTouchStart: TouchEventHandler<HTMLDivElement> = (e) => {/* ... */}

  const onTouchMove: TouchEventHandler<HTMLDivElement> = (e) => {/* ... */}

  const onTouchEnd: TouchEventHandler<HTMLDivElement> = (/* e */) => {/* ... */}

  return (
    <div
      onTouchStart={onTouchStart}
      onTouchMove={onTouchMove}
      onTouchEnd={onTouchEnd}
      style={itemHeightVarStyle}
      className={cn(
        'date-picker-wrapper h-[var(--total-height)] overflow-clip relative',
        className
      )}>
      <div
        className={cn(
          'highlight-select-box h-[var(--date-item-height)] w-full',
          'absolute top-[calc(50%-var(--half-date-item-height))] left-0',
          'border b-solid b-red'
        )}></div>
      <div className="picker-positioner w-full absolute top-[calc(50%-var(--half-date-item-height))]">
        <animated.ol
          style={spring}
          className="year-picker text-center">
          {yearList.map((year) => (
            <li
              key={year}
              className="h-[var(--date-item-height)] leading-[var(--date-item-height)]">
              {year}
            </li>
          ))}
        </animated.ol>
      </div>
    </div>
  )
}
