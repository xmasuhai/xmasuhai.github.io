class Api::V1::MesController < ApplicationController
  # 获取当前用户
  def show
    # get jwt from header
    authorization = request.headers["Authorization"]
    # 去除Bearer
    jwt = authorization.split(" ")[1] rescue ""
    # decode jwt
    payload = JWT.decode jwt, Rails.application.credentials.hmac_secret,
      true, { algorithm: "HS256" } rescue nil
    return render status: :unauthorized, json: { errors: "Invalid JWT" } if payload.nil?
    user_id = payload[0]["user_id"] rescue nil
    currentUser = User.find user_id
    return render status: :unauthorized, json: { errors: "Invalid JWT" } if currentUser.nil?
    render json: { resource: currentUser }
  end
end
  • 会在获取当前用户信息时，读取 header 中的 Authorization 字段，并获取 JWT
  • 再从 JWT 中解码出 user_id 根据 user_id 查询用户信息
  • 以上过程几乎需要在每一个接口重复执行，因此可以创建一个中间件，将这部分代码提取出来
  • 在每个接口中只调用中间件，中间件中再处理这部分逻辑
  • 例如在进入 show 方法前（或进入当前 Controller 前）执行中间件，中间件中再处理这部分逻辑

  • 中间件统一提取 user_id，后续逻辑可直接访问 currentUser 进行查询

class AutoJwt
  # 初始化中间件.
  # @param [Object] app Param documentation.
  # @return [Object]
  def initialize(app)
    @app = app
  end

  # 从请求头中获取 jwt，并解析.
  # @param [Object] env Param documentation.
  # @raise [StandardError]
  # @return [Array]
  def call(env)
    header = env["HTTP_AUTHORIZATION"]

    p("AutoJwt_______________________")
    p env["REQUEST_METHOD"]
    p env["HTTP_AUTHORIZATION"]
    p header
    p("_______________________AutoJwt")

    jwt = header.split(" ")[1] rescue ""
    payload = JWT.decode jwt, Rails.application.credentials.hmac_secret,
      true, { algorithm: "HS256" }
    env["current_user_id"] = payload[0]["user_id"] rescue nil
  end
end
  • 获取请求头 header = env["HTTP_AUTHORIZATION"]
  • 去除 Bearer 得到 jwt 字符串 jwt = header.split(" ")[1] rescue ""
  • 解码 jwt 取出 payload：JWT.decode(...)
  • 获取 user_id 属性放到 env 中（可写）
  • 执行 Controller：@status, @headers, @response = @app.call(env)
    • @status 未明确写出即默认 200
    • @headers 未写出就是默认
    • @response 就是返回的 render json: {...}
  • 默认返回 [ @status, @headers, @response ]
  • 每个中间件都可以获取上一个中间件返回的 [ @status, @headers, @response ]，经过处理后返回

class Api::V1::ItemsController < ApplicationController
  # 分页查询账目.
  #
  # @return [Object]
  def index
    # item = Item.page(params[:page]).per(100)
    items = Item
      .where({ created_at: params[:created_after]..params[:created_before] })
      .page params[:page]
    render json: {
      resources: items,
      pager: {
        page: params[:page],
        per_page: 100,
        count: Item.count
      }
    }
  end

  # 创建账目.
  #
  # @return [Object]
  def create
    item = Item.new amount: params[:amount]
    if item.save
      render json: { resource: item }
    else
      render json: { error: item.errors }
    end
  end
end
  • Ruby 中时间日期范围写法 A..B，例如：{created_at: params[:created_after]..params[:created_before] }

重构 models/user，自动生成 jwt 代替从请求获取 ⇧

  • 之所以获取 headersWithJwt = sign_in user 的逻辑，是因为需要触发sessions_controller#create中登录鉴权的逻辑：
    • payload = { user_id: user.id, email: user.email }
    • hmac_secret = Rails.application.credentials.hmac_secret
    • token = JWT.encode(payload, hmac_secret, "HS256")
    • render status: :ok, json: { jwt: token }
  • 可以改为在 app/models/user.rb 使用方法自己生成这个 jwt 字符串，可以跳过发送请求的逻辑

class User < ApplicationRecord
  validates :email, presence: true

  # User 自己生成 JWT.
  # @return [String]
  def generate_jwt
    payload = { user_id: self.id, email: self.email }
    hmac_secret = Rails.application.credentials.hmac_secret
    JWT.encode(payload, hmac_secret, "HS256")
  end
end
  • 原则就是尽量简化 Controller 中的逻辑，把逻辑放到 model 中

require "jwt"

class Api::V1::SessionsController < ApplicationController
  # 创建会话
  # @return [String] jwt
  # @return [String] errors
  # @return [Integer] status
  def create
    # 如果是测试环境，则跳过验证码查找
    if Rails.env.test?
      return render status: :unauthorized, json: { errors: "邮箱或验证码错误" } if
        params[:code] != "123456"
    else
      canSignIn = ValidationCode.exists? email: params[:email],
        code: params[:code], used_at: nil
      return render status: :unauthorized, json: { errors: "邮箱或验证码错误" } unless canSignIn
    end
    user = User.find_by_email params[:email]
    if user.nil?
      render status: :not_found, json: { errors: "用户不存在" }
    else
      render status: :ok, json: { jwt: user.generate_jwt }
    end
  end
end
  • 使用 render status: :ok, json: { jwt: user.generate_jwt }
  • 在测试用例中，就没有必要再发送post '/api/v1/sessions' 请求获取jwt
  • 可以直接在 user.generate_jwt 中生成 jwt
    • get '/api/v1/items',
    • headers: { "Authorization": "Bearer #{user1.generate_jwt}" }
    • 代替原来 headersWithJwt = sign_in user1

class User < ApplicationRecord
  validates :email, presence: true

  # User 自己生成 JWT.
  # @return [String]
  def generate_jwt
    payload = { user_id: self.id, email: self.email }
    hmac_secret = Rails.application.credentials.hmac_secret
    JWT.encode(payload, hmac_secret, "HS256")
  end

  # 生成 JWT 的 HTTP 头.
  # @return [Hash]
  def generate_auth_header
    { "Authorization": "Bearer #{self.generate_jwt}" }
  end
end
  • 在User模型中新增generate_auth_header 方法，返回包含 Bearer token 的 Authorization 头
  • 这样就可以直接替代 headersWithJwt = sign_in user 来生成 headers: user.generate_auth_header

require_relative '../support/rspec_api_documentation'

resource "账目" do
  get "/api/v1/items" do
    # 基础验证
    authentication :basic, :auth
    # 添加输入
    parameter :page, "页码"
    parameter :created_after, "创建起始时间（筛选条件）"
    parameter :created_before, "创建结束时间（筛选条件）"
    # 响应描述
    # response_field :id, "ID", scope: :resource
    # response_field :amount, "金额（单位：分）", scope: :resource
    # 共有选项语法 with_options
    with_options scope: :resources do
      response_field :id, "ID"
      response_field :amount, "金额（单位：分）"
    end

    # 传参声明示例
    let(:created_after) { "2026-11-10" }
    let(:created_before) { "2026-12-11" }
    let(:current_user) { User.create email: '1@163.com' }
    let(:auth) { "Bearer #{current_user.generate_jwt}" }

    example "获取账目" do
      11.times do
        Item.create amount: 100, created_at: "2026-11-20", user_id: current_user.id
      end
      do_request
      expect(status).to eq 200
      json = JSON.parse response_body
      expect(json["resources"].size).to eq 10
    end
  end
end
  • 验证方式声明 authentication :basic, :auth，默认会在 do_request前添加验证
  • 变量 :auth 的值在下方声明：let(:auth) { "Bearer #{current_user.generate_jwt}" }
  • 变量 :current_user 可提前声明 let(:current_user){ User.create email: '1@163.com' }
  • 后续请求时候，do_request 会自动添加 headers，无需其他额外操作，简化账目接口测试的认证方式
