import { cn } from 'cn'
import * as echarts from 'echarts'
import { useEffect, useId, useRef } from 'react'
import type { EChartsOption } from 'echarts'

interface Props {
  className?: string
  chartData?: { x: number | string; y: number }[]
}

export const DateAmountChart: React.FC<Props> = ({ className, chartData }) => {
  const chartDomRef = useRef<HTMLDivElement>(null)
  const xItems = chartData?.map((item) => item.x)
  const yItems = chartData?.map((item) => item.y)

  useEffect(() => {
    if (!chartDomRef) {
      return
    }
    // 1. 检查该 DOM 是否已经初始化过 ECharts 实例
    let chartInstance = echarts.getInstanceByDom(chartDomRef.current)
    // 2. 如果没有，则进行初始化
    if (!chartInstance) {
      chartInstance = echarts.init(chartDomRef.current)
    }

    const option: EChartsOption = {
      grid: { left: '5%', right: '5%', top: '20%', bottom: '20%' },
      xAxis: {
        type: 'category',
        data: xItems,
        axisLabel: {
          formatter: (label: string /* index: number */) => {
            const sliceSignIndex = label.indexOf('-')
            return label.slice(sliceSignIndex + 1)
          },
        },
      },
      yAxis: {
        type: 'value',
        axisLabel: {
          show: false,
        },
      },
      series: [
        {
          name: '销量',
          type: 'line',
          data: yItems,
          emphasis: {
            focus: 'series',
            scale: 2,
            blurScope: 'series',
            itemStyle: {
              borderWidth: 3,
            },
          },
        },
      ],
    }
    // 3. 渲染你的图表数据
    chartInstance.setOption(option)
    // 4. 组件卸载或重新执行 effect 前销毁实例
    return () => {
      chartInstance?.dispose()
    }
  }, [])

  return <div ref={chartDomRef} className={cn(className)} />
}
  • 外部数据参数 chartData?: { x: number | string; y: number }[]
    • x 轴数据 const xItems = chartData?.map((item) => item.x)
    • y 轴数据 const yItems = chartData?.map((item) => item.y)
  • 声明图表节点 const chartDomRef = useRef<HTMLDivElement>(null)
  • 检查该 DOM 是否已经初始化过 ECharts 实例 echarts.getInstanceByDom(chartDomRef.current)
  • 初始化 chartInstance = echarts.init(chartDomRef.current)
  • 渲染 chartInstance.setOption(option)
  • 组件卸载或重新执行 effect 前销毁实例 return chartInstance?.dispose

// import...

interface Props {
  className?: string
  chartData?: { x: number | string; y: number }[]
}

export const DateAmountChart: React.FC<Props> = ({ className, chartData }) => {
  const chartId = useId()
  const chartDomRef = useRef<HTMLDivElement>(null)
  const chartInstance = useRef<echarts.ECharts | null>(null)

  // 1. 仅负责在挂载时初始化，卸载时销毁
  useEffect(() => {
    if (!chartDomRef.current) {
      return
    }
    // 优先获取已存在的实例，避免重复初始化
    // 1. 检查该 DOM 是否已经初始化过 ECharts 实例
    // 2. 如果没有，则进行初始化
    chartInstance.current = echarts.getInstanceByDom(chartDomRef.current) ??
      echarts.init(chartDomRef.current)

    // 组件卸载或重新执行 effect 前销毁实例
    return () => {
      chartInstance.current?.dispose()
      chartInstance.current = null
    }
  }, [])

  // 2. 负责监听 chartData 变化重新渲染图表
  useEffect(() => {
    const xItems = chartData?.map((item) => item.x)
    const yItems = chartData?.map((item) => item.y)
    // 如果组件已经卸载，则不执行下一步渲染，防止严格模式（Strict Mode） 触发组件双重渲染
    if (!chartInstance.current || chartInstance.current?.isDisposed()) {
      return
    }

    const option: EChartsOption = {
      grid: { left: 0, right: 0, top: '20%', bottom: '20%' },
      tooltip: {
        show: true,
        confine: true,
        // 改用 axis 坐标轴触发，主要在柱状图，折线图等会使用类目轴的图表中使用
        trigger: 'axis',
        formatter: (params: TooltipComponentFormatterCallbackParams) => {
          if (!Array.isArray(params) || params.length === 0) {
            return ''
          }
          const { seriesName, marker, name, data } = params[0]
          const partsDate = name.split('-')
          const label = `${partsDate[0]}年${partsDate[1]}月${partsDate[2]}日`
          const dataValue = data === null ? '无数据' : `¥ ${data}元`
          return `
                <div>
                  <div>
                    ${marker} ${seriesName}
                  </div>
                  <div>
                    <span>${label}</span>
                    <span>${dataValue}</span>
                  </div>
                </div>
              `
        },
      },
      xAxis: {
        type: 'category',
        data: xItems,
        axisLabel: {
          formatter: (label: string /* index: number */) => {
            const sliceSignIndex = label.indexOf('-')
            return label.slice(sliceSignIndex + 1)
          },
        },
      },
      yAxis: {
        type: 'value',
        axisLabel: {
          show: false,
        },
      },
      series: [
        {
          name: '销量',
          type: 'line',
          data: yItems,
          emphasis: {
            focus: 'series',
            scale: 2,
            blurScope: 'series',
            itemStyle: {
              borderWidth: 3,
            },
          },
        },
      ],
    }

    // 3. 渲染图表数据
    chartInstance.current.setOption(option)

    // 记录定时器 ID，并在回调中再次检查实例状态
    const timer = setTimeout(() => {
      if (chartInstance.current?.isDisposed() || xItems?.length === 0) {
        return
      }
      chartInstance.current?.dispatchAction({
        type: 'showTip',
        // 轴触发类型（trigger: 'axis'）时使用:
        seriesIndex: 0, // 第几个 series（通常为 0）
        dataIndex: 0, // 显示第一项（索引从 0 开始）
      })
    }, 800)

    return () => clearTimeout(timer)
  }, [chartData])

  return <div ref={chartDomRef} id={chartId} className={cn(className)} />
}
  • 第一个 useEffect 负责在挂载时初始化，卸载时销毁
    • 避免重复初始化
    • 1. 检查该 DOM 是否已经初始化过 ECharts 实例
    • 2. 如果没有，则进行初始化
    • 组件卸载或重新执行 effect 前销毁实例
  • 第二个 useEffect 负责监听 chartData 变化重新渲染图表
    • 如果组件已经卸载，则不执行下一步渲染，防止严格模式（Strict Mode） 触发组件双重渲染
    • 构造 option
    • 注意 tooltip选项，顶层的 option.tooltip 才支持 trigger: 'axis' 和对应的 formatter
    • 当顶层设置为 axis 时，ECharts 会完全忽略 series 内部的 tooltip 配置
    • 渲染图表数据

import { useEffect, useRef } from 'react'
import type { ECharts } from 'echarts'

interface UseChartCarouselOptions {
  /** ECharts 实例的 Ref */
  chartInstance: React.MutableRefObject<ECharts | null>
  /** 图表容器 DOM 的 Ref（用于绑定移动端触摸事件） */
  chartDomRef: React.RefObject<HTMLDivElement | null>
  /** 数据总长度（数据更新或为 0 时自动重置轮播） */
  dataLength: number
  /** 轮播间隔时间（毫秒），默认 2000ms */
  interval?: number
  /** 初始化延迟触发时间（毫秒），默认 800ms（等待图表动画完成） */
  initDelay?: number
  /** 目标系列索引，默认 0 */
  seriesIndex?: number
  /** 是否开启自动轮播，默认 true */
  enabled?: boolean
}

export function useChartTooltipCarousel({
  chartInstance,
  chartDomRef,
  dataLength = 0,
  interval = 2000,
  initDelay = 800,
  seriesIndex = 0,
  enabled = true,
}: UseChartCarouselOptions) {
  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null)
  const touchEndTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null)
  const currentCarouselIndexRef = useRef<number>(0)

  useEffect(() => {
    const chart = chartInstance.current
    const dom = chartDomRef.current

    // 防御性检查：实例未挂载、已销毁、未启用或无数据时不启动
    if (!enabled || !chart || chart.isDisposed() || !dom || dataLength <= 0) {
      return
    }

    // 1. 触发指定索引的 Tooltip 与高亮状态
    const showTipAt = (index: number) => {
      if (chart.isDisposed()) return
      chart.dispatchAction({
        type: 'showTip',
        seriesIndex,
        dataIndex: index,
      })
      chart.dispatchAction({ type: 'downplay', seriesIndex })
      chart.dispatchAction({
        type: 'highlight',
        seriesIndex,
        dataIndex: index,
      })
    }

    // 2. 启动与停止定时器
    const stopCarousel = () => {
      if (timerRef.current) {
        clearInterval(timerRef.current)
        timerRef.current = null
      }
      if (touchEndTimerRef.current) {
        clearTimeout(touchEndTimerRef.current)
        touchEndTimerRef.current = null
      }
    }

    const startCarousel = () => {
      stopCarousel()
      // 只有 1 条数据时展示第一项即可，无需开启循环定时器
      if (dataLength <= 1) return

      timerRef.current = setInterval(() => {
        // 轮播逻辑精髓 取余 实现循环
        currentCarouselIndexRef.current = (currentCarouselIndexRef.current + 1) % dataLength
        showTipAt(currentCarouselIndexRef.current)
      }, interval)
    }

    // 3. PC 端鼠标交互
    const handleMouseOver = (params: { dataIndex?: number }) => {
      stopCarousel()
      if (typeof params.dataIndex === 'number') {
        currentCarouselIndexRef.current = params.dataIndex
      }
    }
    const handleGlobalOut = () => {
      startCarousel()
    }

    chart.on('mouseover', handleMouseOver)
    chart.on('globalout', handleGlobalOut)

    // 4. 移动端触摸交互
    const handleTouchStart = () => {
      stopCarousel()
    }
    const handleTouchEnd = () => {
      // 延迟 500ms 恢复，避免手指离开瞬间的点击事件与轮播冲突
      touchEndTimerRef.current = setTimeout(() => {
        startCarousel()
      }, 500)
    }

    dom.addEventListener('touchstart', handleTouchStart, { passive: true })
    dom.addEventListener('touchend', handleTouchEnd)

    // 5. 页面切后台/锁屏时自动暂停，节省移动端性能
    const handleVisibilityChange = () => {
        document.hidden ? stopCarousel() : startCarousel();
    }
    document.addEventListener('visibilitychange', handleVisibilityChange)

    // 6. 首次延迟展示第一项并启动轮播
    const initTimer = setTimeout(() => {
      if (chart.isDisposed()) return
      currentCarouselIndexRef.current = 0
      showTipAt(0)
      startCarousel()
    }, initDelay)

    // 7. 清理所有副作用
    return () => {
      clearTimeout(initTimer)
      stopCarousel()
      document.removeEventListener('visibilitychange', handleVisibilityChange)
      dom.removeEventListener('touchstart', handleTouchStart)
      dom.removeEventListener('touchend', handleTouchEnd)
      if (!chart.isDisposed()) {
        chart.off('mouseover', handleMouseOver)
        chart.off('globalout', handleGlobalOut)
      }
    }
  }, [chartInstance, chartDomRef, dataLength, interval, initDelay, seriesIndex, enabled])
}

interface Props<T extends FormItemType, U extends string | number = string> {
  className?: string
  formItemType: T
  formItemTypeProps: FormItemTypeProps<T, U>
}

export function FormItem<T extends FormItemType, U extends string | number = string>(
  props: Props<T, U>
) {
  const { className, formItemType, formItemTypeProps } = props
  const renderContent = () => {
    // TS 在 switch (formItemType) 中不会把泛型 T 缩窄为字面量
    // 因此 formItemTypeProps 仍为联合类型
    // 展开到各具体组件时 value 等属性类型不兼容
    // 在各分支断言为对应字面量类型的 props
    switch (formItemType) {
      case 'input':
        return (
          <InputText
            type={formItemType}
            {...(formItemTypeProps as FormItemTypeProps<'input'>)}
            className={className}
          />
        )
      case 'emoji':
        return (
          <InputEmoji
            type={formItemType}
            {...(formItemTypeProps as FormItemTypeProps<'emoji'>)}
            className={className}
          />
        )
      case 'validateCodeSend':
        return (
          <InputValidateCodeSend
            type={formItemType}
            {...(formItemTypeProps as FormItemTypeProps<'validateCodeSend'>)}
            className={className}
          />
        )
      case 'optionSelector':
        return (
          <OptionSelector<U>
            type={formItemType}
            {...(formItemTypeProps as FormItemTypeProps<'optionSelector', U>)}
            className={className}
          />
        )
      default:
        return null
    }
  }
  return <>{renderContent()}</>
}
  • 泛型可选参数需要给默认类型 interface Props<T extends FormItemType, U extends string | number = string>
  • 类型传参 formItemTypeProps: FormItemTypeProps<T, U>
  • case 'optionSelector': 收窄类型
    • <OptionSelector<U> type={formItemType} {...(formItemTypeProps as FormItemTypeProps<'optionSelector', U>)} .../>

import type { SelectorOption } from '@/components/form/item/FormItemType'
import { cn } from 'cn'

interface Props<T> {
  // 仅用作区分 FormItemType 类型
  type?: 'optionSelector'
  className?: string
  // for 属性，用于内部 input 标签 id 属性联动
  labelFor?: string
  label?: string
  hasLabelColon?: boolean
  ifOneLineLabel?: boolean
  errorHint?: string
  hasErrorPlaceholder?: boolean
  value?: T
  options?: SelectorOption<T>[]
  placeholder?: string
  onChange?: (value: T) => void
  onBlur?: () => void
  onFocus?: () => void
  maxWidth?: string
}

export const OptionSelector = <T extends string | number>({
  className,
  label,
  ifOneLineLabel = false,
  labelFor,
  hasLabelColon = true,
  errorHint,
  hasErrorPlaceholder = true,
  placeholder,
  value,
  options,
  onChange,
  onBlur,
  onFocus,
  maxWidth = '12em',
}: Props<T>) => {
  const labelForId = labelFor && `form-item-select-${labelFor}`
  const labelChildren = (
    <>
      {label}
      {hasLabelColon ? '：' : ''}
    </>
  )
  const labelTag = (
    <label {...(labelForId ? { htmlFor: labelForId } : {})} className="form-label-sc">
      {labelChildren}
    </label>
  )
  const handleChange = (e: React.ChangeEvent<HTMLSelectElement>) => {
    onChange?.(e.target.value as T)
  }
  return (
    <section style={{ maxWidth }} className={cn('form-item-sc', className)}>
      <div className={cn('form-item-content-sc', ifOneLineLabel ? 'flex-row' : 'flex-col')}>
        <span className="flex items-center">{label && labelTag}</span>
        <select
          id={labelForId}
          value={value}
          onChange={handleChange}
          onBlur={onBlur}
          onFocus={onFocus}
          className={cn(
            'where:(text-1em)',
            'grow-1 min-h-[var(--input-min-height)] p-x-.5em',
            'b-1 rounded-[var(--input-radius)] shadow-[inset_0_0_3px_var(--input-shadow)]',
            errorHint ? 'b-[var(--error-color)]' : 'b-[var(--input-border-color)]'
          )}
          placeholder={placeholder}>
          {options?.map((option) => (
            <option key={option.value} value={option.value}>
              {option.label}
            </option>
          ))}
        </select>
      </div>
      {(errorHint || hasErrorPlaceholder) && (
        <span className="form-item-error-sc w-fit buzz">{errorHint ?? <>&nbsp;</>}</span>
      )}
    </section>
  )
}
