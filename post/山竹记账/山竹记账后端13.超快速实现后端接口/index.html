    • sign:string
    • deleted_at:datetime
    • created_at:datetime
    • updated_at:datetime

class CreateTags < ActiveRecord::Migration[7.2]
  def change
    create_table :tags do |t|
      t.references :user, null: false, foreign_key: false
      t.string :name
      t.string :sign
      t.datetime :deleted_at

      t.timestamps
    end
  end
end
  • t.references :user, null: false, foreign_key: false 表示引用到 user 表，不能为空，不使用外键（不方便测试）
  • t.string :name, null: false 不能为空，如果此处不写 null: false，那么需要在 ruby 验证器中添加不能为空的验证

  • 目前 db/schema.rb 中还没有这个表需要创建，运行 bin/rails db:migrate

class Api::V1::TagsController < ApplicationController
  # 获取标签.
  # @return [Object]
  def index
    current_user = User.find request.env["current_user_id"]
    return render status :not_found if current_user.nil?

      .where(user_id: current_user.id)
      .page(params[:page])
    render json: { resources: tags }
  end
end

class Api::V1::TagsController < ApplicationController
  # 获取标签.
  # @return [Object]
  def index
    current_user = User.find request.env["current_user_id"]
    return render status :not_found if current_user.nil?
    tags = Tag
      .where(user_id: current_user.id)
      .page(params[:page])

    render json: {
      resources: tags,
      pager: {
        page: params[:page] || 1,
        per_page: Tag.default_per_page,
        count: Tag.count
      }
    }
  end
end
