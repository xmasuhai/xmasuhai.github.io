export const ItemTab = defineComponent({
  name: 'ItemTab',
  props: {
    tabName: {
      type: String as PropType<string>,
      required: true,
    }
  },
  setup( _props, _ctx ) {
    return () => (
      <div>{_ctx.slots?.default?.()}</div>
    )
  },
})
  • SwitchTabs 通过 _ctx.slots?.default?.() 获取子组件，相当于 react 的 children
    • 插槽内容可能为多个VNode元素的数组
    • 打印出查看

  • 可以对比 _ctx.slots?.default?.()?.[0]?.type === ItemTab 来判断子组件类型是否为 ItemTab
  • 遍历判断数组中所有 VNode 元素的 type 是否为 ItemTab，如果不是则抛出异常
  • 注意插件 _ctx.slots?.default?.() 必须写在 setup 的返回函数中，否则会报警告：

    • Invoke the slot function inside the render function instead.

4. 实现 Tabs 功能 ⇧

  • SwitchTabs 通过默认插槽获取子组件 ItemTab，并通过 props 传入 selected 属性，来控制当前选中的 tab
  • props.selected 和 遍历的 slot.props?.tabName 进行比较，来判断当前 tab 是否被选中，添加 selected 样式
  • 添加 onUpdateSelected 事件处理函数，更新 selected 的值，来实现 tab 切换功能
    • 通过 props 中的 onUpdateSelected 回调函数，执行时传入当前 tab 的名称，来更新父组件的状态

import { defineComponent, type PropType } from 'vue'
import { ItemTab } from '@/components/tab/ItemTab.tsx'
import { cn } from 'cn'

export const SwitchTabs = defineComponent({
  name: 'SwitchTabs',
  props: {
    selected: {
      type: String as PropType<string>,
    },
    onUpdateSeleted: {
      type: Function as PropType<(name: string) => void>,
    },
  },
  setup( _props, _ctx ) {
    return () => {
      const defaultSlotsArray = _ctx.slots?.default?.() ?? []
      const ifAllItemTab = defaultSlotsArray
        .every(slot => slot.type === ItemTab &&
          !!slot.props?.tabName)
      if (!ifAllItemTab) {
        throw new Error(`SwitchTabs: 子组件必须是 ItemTab，且必须有 tabName 属性`)
      }
      return (
        <>
          {/* 切换标签 */}
          <nav>
            <menu>
              {defaultSlotsArray
                .map((slot, index) => (
                  <li
                    key={index + slot?.props?.tabName}
                    class={cn(
                      _props.selected === slot?.props?.tabName ? 'bg-red' : ''
                    )}
                    onClick={() => {_props.onUpdateSeleted?.(slot?.props?.tabName)}}>
                    {slot?.props?.tabName}
                  </li>
                ))}
            </menu>
          </nav>

          {/* 标签内容 */}

        </>
      )
    }
  },
})

export const SwitchTabs = defineComponent({
  name: 'SwitchTabs',
  props: {
    selected: {
      type: String as PropType<string>,
    },
  },
  setup(_props, _ctx) {
    return () => {
      const defaultSlotsArray = _ctx.slots?.default?.() ?? []
      const ifAllItemTab = defaultSlotsArray.every(
        (slot) => slot.type === ItemTab && !!slot.props?.tabName,
      )
      if (!ifAllItemTab) {
        throw new Error(`SwitchTabs: 子组件必须是 ItemTab，且必须有 tabName 属性`)
      }

      return (
        <>
          {/* 切换标签 */}
          <nav>
            <menu>
              {defaultSlotsArray.map((slot, index) => (
                <li
                  key={index + slot?.props?.tabName}
                  class={cn(_props.selected === slot?.props?.tabName ? 'bg-red' : '')}
                  onClick={() => {
                    _ctx.emit('update:selected', slot?.props?.tabName)
                  }}>
                  {slot?.props?.tabName}
                </li>
              ))}
            </menu>
          </nav>

          {/* 标签内容 */}
          <section>
            {defaultSlotsArray
              .find((slot) => slot.props?.tabName === _props.selected)}
          </section>
        </>
      )
    }
  },
})

import { defineComponent, type PropType } from 'vue'
import { ItemTab } from '@/components/tab/ItemTab.tsx'
import { cn } from 'cn'

export const SwitchTabs = defineComponent({
  name: 'SwitchTabs',
  props: {
    selected: {
      type: String as PropType<string>,
    },
  },
  setup(_props, _ctx) {
    const selectedClass = (tabName: string) => _props.selected === tabName
      ? cn(
        'relative',
        `after:(content-[""] block absolute bottom-0 left-0 w-full h-4px bg-[var(--tabs-indicator-bg)])`,
      )
      : ''

    return () => {
      const defaultSlotTabs = _ctx.slots?.default?.() ?? []
      const ifAllItemTab = defaultSlotTabs.every(
        (slot) => slot.type === ItemTab && !!slot.props?.tabName,
      )
      if (!ifAllItemTab) {
        throw new Error(`SwitchTabs: 子组件必须是 ItemTab，且必须有 tabName 属性`)
      }

      return (
        <>
          {/* 切换标签 */}
          <nav>
            <menu class="flex justify-between items-center text-color-[var(--navbar-text-color)]">
              {defaultSlotTabs.map((slot, index) => (
                <li
                  key={index + slot?.props?.tabName}
                  class={cn(
                    'flex-grow-1 flex-shrink-0 text-center',
                    'p-y-12px bg-[var(--navbar-bg-end)]',
                    selectedClass(slot?.props?.tabName)
                  )}
                  onClick={() => {
                    _ctx.emit('update:selected', slot?.props?.tabName)
                  }}>
                  {slot?.props?.tabName}
                </li>
              ))}
            </menu>
          </nav>

          {/* 标签内容 */}
          <section>
            {defaultSlotTabs
              .find((slot) => slot.props?.tabName === _props.selected)}
          </section>
        </>
      )
    }
  },
})
  • 使用 unocss 的 transformerVariantGroup 配置 来表示 after:() 伪元素语法中多个样式
  • 给选中的 tab 添加下划线样式 after:(content-[""] block absolute bottom-0 left-0 w-full h-4px bg-[var(--tabs-indicator-bg)])
