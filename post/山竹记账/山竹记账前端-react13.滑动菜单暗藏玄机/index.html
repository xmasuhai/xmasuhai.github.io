import { MenuList } from "@/components/menu/MenuList"
import { OverlayMask } from "@/components/menu/OverlayMask"
import { UserHeader } from "@/components/menu/UserHeader"
import { useMenuStore } from "@/stores/useMenuStore"
import { animated, useSpring } from "@react-spring/web"
import { cn } from "cn"

interface Props {}

export const FloatingOverlayMenu: React.FC<Props> = () => {
  const { visible, } = useMenuStore()
  const menuStyle = useSpring({
    transform: visible ? 'translateX(0%)' : 'translateX(-100%)',
    opacity: visible ? 1 : 0,
  })
  return (
    <>
      <animated.div
        style={menuStyle}
        className={cn(
          'floating-menu',
          'fixed top-0 left-0',
          'w-70vw max-w-20em h-screen',
          'flex flex-col',
          'z-[var(--z-menu)]',
        )}>
        <UserHeader className="grow-0 shrink-0" />
        <MenuList className="grow-1 shrink-1" />
      </animated.div>

      <OverlayMask />
    </>
  )
}
  • useSpring(Object) 中接收除了接口默认的配置参数外，
    • 还可以接收动画相关的属性，例如 opacity、transform、visibility 等
    • 返回值即可以响应式的 style 对象值 SpringValues，可在动画配置事件中打印查看
  • 注意移动过渡动画 transform: visible ? 'translateX(0%)' : 'translateX(-100%)', 中不可写成 translateX(0)
    • translateX(0) 没有补间动画
  • 透明度渐变 opacity: visible ? 1 : 0,

import { useMenuStore } from "@/stores/useMenuStore"
import { animated, useSpring } from "@react-spring/web"
import { cn } from "cn"

interface Props { }

export const OverlayMask: React.FC<Props> = () => {
  const { visible, setVisible, } = useMenuStore()
  const [maskVisible, setMaskVisible] = useState(false)
  const maskAnimeStyle = useSpring({
    opacity: visible ? 1 : 0,
    visibility: maskVisible ? 'visible' : 'hidden',
    config: { duration: 1000,},
    onStart(result) {
      console.log('maskAnimeStyle_______________________')
      console.log('%c%s', 'color: #aa00ff', visible);
      console.log('%c⧭', 'color: #00a3cc', result);
      console.log('%c⧭', 'color: #00a3cc', result.value);
      console.log('_______________________maskAnimeStyle')
      if (result.value.opacity < 0.1) {
        setMaskVisible(true)
      }
    },
    onRest({ value }) {
      if (value.opacity < 0.1) {
        setMaskVisible(false)
      }
    },
  })
  return (
    <animated.div
      style={maskAnimeStyle}
      onClick={() => setVisible(false)}
      className={cn(
        'overlay-mask',
        'fixed top-0 left-0 w-screen h-screen',
        'bg-[var(--overlay-mask-bg)] z-[var(--z-index-float-overlay)]',
      )}>
    </animated.div>
  )
}
  • 为了不遮挡可点击页面元素，需要将遮罩层初始设为不可见
    • visibility: visible ? 'visible' : 'hidden'
  • visible 变化重新计算节点，对应的 maskAnimeStyle 也会变化
  • 在 onStart 钩子中
    • 打开动画 opacity: 从 0 -> 1
    • 关闭动画 opacity: 从 1 -> 0
    • 但是 result.value.opacity 中的数值打印出来并不是完全等于 0
    • 可以放宽松地和 0.1 比较，符合条件就 setMaskVisible(true)
  • 在 onRest 钩子中
    • 打开动画 opacity: 从 0 -> 1
    • 关闭动画 opacity: 从 1 -> 0
    • 同样地也还是判断 value.opacity < 0.1

  • 之后发现和 visible 状态的逻辑行为一致，可直接使用 visible 代替 maskVisible 即可
  • 直接都放到 useSpring({}) 的配置样式中即可，不用单独再设置 setMaskVisible

5. 插曲：一个你可能理解不了的bug ⇧

当跳转到其他路由后，useMenuStore 状态中 的 visible 是否还在内存中

  • 仍然存在
  • 使用 <Link> 或者 <NavLink>；并且使用 createBrowserRouter 的 history API，并不会刷新页面
  • 不刷新页面意味着内存不会被刷新，跳转另外路由时，visible 状态会保持不变
  • 此时再点击回退，由于组件内的状态并不在状态管理，会随组件的销毁而删除
  • 为了避免不更新状态的bug，可以将状态都放到状态管理文件中
  • 或者使得组件内的状态初始值依赖于状态值即可
