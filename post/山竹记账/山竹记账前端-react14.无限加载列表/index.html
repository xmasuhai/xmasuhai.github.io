1. 无限加载的实现思路 ⇧

  • 访问 /items 页面，请求 /items?page=1
  • 每次加载10项数据，返回 { resources: [], pager: {count: 10, per_page: 10, page: 1, }, }
  • 点击加载更多时，在原先数据后再插入10项数据 ?page=2
  • 页码乘以每页项目数，和总条目数对比，判断是否有更多数据加载
  • 使用 hasMore = (page - 1) * per_page + resourses.length < count 计算 是否还有更多数据
  • 报错和特殊情况情况
    • 加载第一页：resources 返回空数组，则显示暂无数据；
    • 请求第一页时，出错了，添加错误提示
    • 请求第二页时，出错了，添加错误提示…
  • 显示正在加载状态，改变“加载更多”的文案为“正在加载”；成功后变为“没有更多”或者“加载更多”

  • 使用 useSWRInfinite 的数据模型，每次获取一页数据，向 data 中添加一份当前页的数据

  • 使用 useSWRInfinite 提供的 setSize API 去实现加载下一页数据
  • size 即当前页码，从 1 开始
  • const loadMore = () => { setSize(size + 1) } 可以将 data 的 size 增加一个
    • const { data, size, setSize, } = useSWRInfinite( getKey, (path) => fetcher(path), )
  • 使用 data.flatMap(({ data: { resources, pager: _pager } }, dataIndex) => ()) 拍平二维数组
    • data = [{resources: [], pager: {}}, {resources: [], pager: {}}, {resources, pager}, ...]
    • 返回合并数组 resources

export const createItemsMockResponse = (
  {
    perPage = 10,
    page = 1,
    count = 10,
  },
  attrs?: Partial<Item>,
) => {
  const sentCount = (page - 1) * perPage
  const rest = count - sentCount
  return {
    code: 0,
    data: {
      resources: rest > 0 ? createMockItemList(Math.min(rest, perPage), attrs) : [],
      pager: {
        page,
        per_page: perPage,
        count,
      },
    },
  }
}

  • 在 createItemsMockResponse 中
    • 判断已发送（上一页）的条目数 sentCount = (page - 1) * perPage
    • 计算剩余条目数 rest = count - sentCount
    • 由 rest > 0 可判断 resources 是否还有更多数据
    • 若 rest > 0，则返回 resources，否则返回空数组
    • Math.min(rest, perPage) 返回的数据条目数不超过 perPage
  • 目前就确保了发送所有数据等于总条目数后，返回的 resources 为空；即不会发送多于 count个数的条目

  • 根据上一次的请求结果来判断

import { Item, Resources, ResponseData } from "@/global"
import { request } from "@/lib/request/request"
import { cn } from "cn"
import useSWRInfinite from "swr/infinite"

export const ItemsList: React.FC<Props> = () => {
  type ResourcesData = ResponseData<Resources<Item>>
  const getKey = (pageIndex: number = 0, prev: ResourcesData) => {
    if (prev) {
      const sentCount = (prev.data.pager.page - 1) * prev.data.pager.per_page + prev.data.resources.length
      const count = prev.data.pager.count
      if (sentCount >= count) { return null}
    }
    return `/api/v1/items?page=${pageIndex + 1}`
  }
  const fetcher = async (path: string) => (await request.get<ResourcesData>(path)).data
  const { data, size, setSize, } = useSWRInfinite(
    getKey,
    (path) => fetcher(path),
  )
  const loadMore = () => { setSize(size + 1) }
  if (!data) {
    return (<span>还没搞定</span>)
  } else {
    return (
      <ol
        className={cn(
          'accounts',
          'flex flex-col overflow-auto scrollbar-hide children:(basis-[78px] shrink-0)',
          '[&>li]:[--sibling-idx:calc(1*sibling-index())]',
          '[&>li>div:first-child]:bg-[oklch(70%_0.15_calc(var(--sibling-idx)*70))]',
        )}>
        {
          data.flatMap(({ data: { resources, pager: _pager } }, dataIndex) => (
            resources.map((item: Item) => (
              <li
                key={`${dataIndex}-${item.id}`}
                className={cn(
                  'account-item',
                  'flex justify-between items-start py-[12px] px-[1em] ',
                  'b-b-[1px] b-t-[1px] b-t-[transparent] b-b-[var(--button-border-color)]',
                )}>
                <div className="sign grow-0 shrink-0 size-[48px] rounded-[50%] flex justify-center items-center my-0 mx-.5em">
                  <span>😀</span>
                </div>

                <div className="mx-.25em grow-1 shrink-1">
                  <div className="flex justify-between">
                    <span className="text-[var(--list-tag-text)]">{item.note}</span>
                    <span className="text-[var(--list-amount-text)]">¥&nbsp;{item.amount / 100}</span>
                  </div>

                  <div className="text-[var(--list-time-text)] mt-.25em">
                    2027-01-01 12:00:00
                  </div>
                </div>
              </li>
            ))))
        }

            加载更多

      </ol>
    )

  • data 来自于 useSWRInfinite
  • 在 useSWRInfinite 的参数 getKey 函数中判断是否存在下一页
    • 如果已请求的条目数大于或等于条目总数，就返回 null，相当于不请求数据了；否则返回请求URL /api/v1/items?page=${pageIndex + 1}
    • 已请求的条目数可由计算公式得出：(prev.data.pager.page - 1) * prev.data.pager.per_page + prev.data.resources.length
    • 条目总数为 prev.data.pager.count
  • 目前实现了请求完最后一页，不会再次请求下一页数据了；即不会请求超出页数之外的数据了
  • 对data数据进行两层的遍历，拍平二维数组
    • 解构出每个数据中的 resources，使用 flatMap 拍平 resources
    • 再遍历生成节点

import { InLoading } from "@/components/InLoading"
import { Item, Resources, ResponseData } from "@/global"
import { request } from "@/lib/request/request"
import { cn } from "cn"
import styled from "styled-components"
import useSWRInfinite from "swr/infinite"

interface Props {
  startDate: string
  endDate: string
}

export const ItemsList: React.FC<Props> = ({
  // startDate,
  // endDate,
}) => {
  type ResourcesData = ResponseData<Resources<Item>>
  // SWR key
  const getKey = (pageIndex: number = 0, prev: ResourcesData) => {
    if (prev) {
      const sentCount = (prev.data.pager.page - 1) * prev.data.pager.per_page + prev.data.resources.length
      const count = prev.data.pager.count
      if (sentCount >= count) { return null }
    }
    return `/api/v1/items?page=${pageIndex + 1}`
  }
  const fetcher = async (path: string) => (await request.get<ResourcesData>(path)).data
  const { data, size, setSize, error, isLoading, isValidating, /*   mutate, */ } = useSWRInfinite(
    getKey,
    (path) => fetcher(path),
  )

  const loadMore = () => { setSize(size + 1) }

  const Div = styled.div`
    padding: 16px;
    text-align: center;
  `
  const errorTip = error && (<Div className="color-[var(--error-color)]">数据加载失败，请稍后刷新页面</Div>)

  const inLoadingState = isLoading || isValidating

  console.log('ItemsSummary_______________________')
  console.log('%c⧭', 'color: #733d00', data);
  console.log('%c⧭ size', 'color: #733d00', size);
  console.log('%c⧭ isValidating', 'color: #733d00', isValidating);
  console.log('%c⧭ isLoading', 'color: #733d00', isLoading);
  console.log('_______________________ItemsSummary')

  if (!data) {
    return (
      <div className="p-1em text-center">
        {errorTip}
        {inLoadingState && (<Div>数据加载中……</Div>) }
      </div>
    )
  } else {
    const lastPageData = data?.[size - 1]
    const { page, per_page, count } = lastPageData?.data.pager ?? {}
    const hasMore = (page - 1) * per_page + lastPageData?.data.resources.length < count

    return (
      <ol
        className={cn(
          'accounts',
          'flex flex-col overflow-auto scrollbar-hide children:(basis-[78px] shrink-0)',
          '[&>li]:[--sibling-idx:calc(1*sibling-index())]',
          '[&>li>div:first-child]:bg-[oklch(70%_0.15_calc(var(--sibling-idx)*70))]',
        )}>
        {
          data.flatMap(({ data: { resources, pager: _pager } }, dataIndex) => (
            resources.map((item: Item) => (
              <li
                key={`${dataIndex}-${item.id}`}
                className={cn(
                  'account-item',
                  'flex justify-between items-start py-[12px] px-[1em] ',
                  'b-b-[1px] b-t-[1px] b-t-[transparent] b-b-[var(--button-border-color)]',
                )}>
                <div className="sign grow-0 shrink-0 size-[48px] rounded-[50%] flex justify-center items-center my-0 mx-.5em">
                  <span>😀</span>
                </div>

                <div className="mx-.25em grow-1 shrink-1">
                  <div className="flex justify-between">
                    <span className="text-[var(--list-tag-text)]">{item.note}</span>
                    <span className="text-[var(--list-amount-text)]">¥&nbsp;{item.amount / 100}</span>
                  </div>

                  <div className="text-[var(--list-time-text)] mt-.25em">
                    2027-01-01 12:00:00
                  </div>
                </div>
              </li>
            ))))
        }

        {errorTip}

        <li className="px-1em pt-2em mb-2em text-center">
          <button
            disabled={!hasMore || inLoadingState}
            onClick={loadMore}
            className={cn(
              'primary-btn flex items-center justify-center',
              hasMore && !inLoadingState ? '' : 'disabled',
            )}>
            {
              !hasMore && !inLoadingState
                ? '没有更多数据了'
                : inLoadingState
                  ? (<InLoading direction="horizontal" hasRainbow={false} className="size-1em flex text-1em fill-white" />)
                  : '加载更多'
            }
          </button>
        </li>
      </ol>
    )

  }

}
