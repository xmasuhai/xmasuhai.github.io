const https = require('https');
const fs = require('fs');

/**
 * 抓取网页 HTML
 */
function fetchHTML(url) {
  return new Promise((resolve, reject) => {
    https.get(url, (res) => {
      if (res.statusCode >= 300 && res.statusCode < 400 && res.headers.location) {
        return resolve(fetchHTML(res.headers.location));
      }
      let data = '';
      res.on('data', (chunk) => { data += chunk; });
      res.on('end', () => resolve(data));
      res.on('error', reject);
    }).on('error', reject);
  });
}

/**
 * 解码 HTML 实体
 */
function decodeHtmlEntities(str) {
  return str
    .replace(/&amp;/g, '&')
    .replace(/&lt;/g, '<')
    .replace(/&gt;/g, '>')
    .replace(/&quot;/g, '"')
    .replace(/&#39;/g, "'")
    .replace(/&apos;/g, "'");
}

/**
 * 去除 HTML 标签，只保留纯文本
 */
function stripHtmlTags(str) {
  return str.replace(/<[^>]+>/g, '').trim();
}

/**
 * 将 "U+1F600" 或 "U+1F636 U+200D U+1F32B U+FE0F" 转为 "\u{1F600}" 或 "\u{1F636}\u{200D}\u{1F32B}\u{FE0F}"
 * 只有一个反斜杠
 */
function convertUnicodeFormat(codeStr) {
  const codePoints = [];
  const regex = /U\+([0-9a-fA-F]+)/g;
  let match;
  while ((match = regex.exec(codeStr)) !== null) {
    codePoints.push(match[1].toUpperCase());
  }
  return codePoints.map(cp => '\\u{' + cp + '}').join('');
}

/**
 * 手动序列化 JSON，确保 unicode 字段中只有一个反斜杠
 * 不使用 JSON.stringify，避免反斜杠被转义
 */
function buildJSON(output) {
  let lines = [];
  lines.push('{');
  lines.push('  "source": ' + JSON.stringify(output.source) + ',');
  lines.push('  "version": ' + JSON.stringify(output.version) + ',');
  lines.push('  "total": ' + output.total + ',');
  lines.push('  "data": [');
  for (let i = 0; i < output.data.length; i++) {
    const e = output.data[i];
    lines.push('    {');
    lines.push('      "category": ' + JSON.stringify(e.category) + ',');
    lines.push('      "num": ' + e.num + ',');
    lines.push('      "code": ' + JSON.stringify(e.code) + ',');
    // unicode 字段直接写入字面量，不经过 JSON.stringify 转义反斜杠
    lines.push('      "unicode": "' + e.unicode + '",');
    lines.push('      "shortName": ' + JSON.stringify(e.shortName));
    lines.push('    }' + (i < output.data.length - 1 ? ',' : ''));
  }
  lines.push('  ]');
  lines.push('}');
  return lines.join('\n');
}

/**
 * 解析 emoji 表格
 */
function parseEmojiTable(html) {
  const results = [];
  let currentCategory = '';

  const trRegex = /<tr[^>]*>([\s\S]*?)<\/tr>/g;
  let trMatch;

  while ((trMatch = trRegex.exec(html)) !== null) {
    const rowContent = trMatch[1];

    // 检查是否为 mediumhead 分类行
    if (rowContent.includes('mediumhead')) {
      const titleMatch = rowContent.match(/<a[^>]*>([\s\S]*?)<\/a>/);
      if (titleMatch) {
        currentCategory = decodeHtmlEntities(titleMatch[1].trim());
      }
      continue;
    }

    // 跳过表头行（th 标签）
    if (/<th[\s>]/.test(rowContent)) continue;

    // 提取所有 td 单元格
    const tds = [];
    const tdRegex = /<td[^>]*>([\s\S]*?)<\/td>/g;
    let tdMatch;
    while ((tdMatch = tdRegex.exec(rowContent)) !== null) {
      tds.push(tdMatch[1]);
    }

    // 数据行应该有 9 列
    if (tds.length < 9) continue;

    // 第一列是序号
    const numText = stripHtmlTags(decodeHtmlEntities(tds[0]));
    if (!numText || !/^\d+$/.test(numText)) continue;
    const num = parseInt(numText, 10);

    // 第二列是 Code (含 <a> 标签)，去除标签后得到 "U+1F600"
    const code = stripHtmlTags(decodeHtmlEntities(tds[1]));

    // CLDR Short Name 在最后一列 (index 8)
    let shortName = stripHtmlTags(decodeHtmlEntities(tds[8]));

    // 去除新增 emoji 标记 ⊛
    shortName = shortName.replace(/^⊛\s*/, '').trim();

    // 转换 unicode 格式
    const unicode = convertUnicodeFormat(code);

    results.push({
      category: currentCategory,
      num: num,
      code: code,
      unicode: unicode,
      shortName: shortName
    });
  }

  return results;
}

  const jsonStr = buildJSON(output);
