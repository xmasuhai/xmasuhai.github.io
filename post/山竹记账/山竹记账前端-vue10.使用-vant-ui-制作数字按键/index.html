1. 制作数字按键 ⇧

  • 声明按钮列表数据，遍历渲染；声明按钮类型
    • 按钮支持文字或图标
  • 使用二维布局 grid 做不规则分布的按钮
  • 四列四行，其中 清空（归零）、删除（图标）、提交、0 位数字占据跨多行/列

  • 父容器设置背景色 #ddd，留出子节点空隙 gap-1px，设置子节点背景为白色，形成类似边框线条
  • 遍历的按钮对象需要设置具体不同的 grid-area-[]，具体网格样式配置可参考：css grid generator

      <>
        <div class="">
          <span class="notes">
            <SvgIcon name="date" />

export const time = (date?: Date) => {
  if(!date) {date = new Date()}
  // 目前支持的格式有 'YYYY-MM-DD HH:mm:ss.sss'
  const api = {
    format: (pattern: string = 'YYYY-MM-DD') => {
      // YYYY
      const year = date.getFullYear()
      // MM
      const month = date.getMonth() + 1
      // DD
      const day = date.getDate()
      // HH
      const hour = date.getHours()
      // mm
      const minute = date.getMinutes()
      // ss
      const second = date.getSeconds()
      // sss
      const millisecond = date.getMilliseconds()

      return pattern
        .replace(/YYYY/g, year.toString())
        // 使用 padStart 长度不足则补零
        .replace(/MM/g, month.toString().padStart(2, '0'))
        .replace(/DD/g, day.toString().padStart(2, '0'))
        .replace(/HH/g, hour.toString().padStart(2, '0'))
        .replace(/mm/g, minute.toString().padStart(2, '0'))
        .replace(/ss/g, second.toString().padStart(2, '0'))
        .replace(/sss/g, millisecond.toString().padStart(3, '0'))
    }
  }

  return {
    api,
  }
}

import { computed, defineComponent, ref } from 'vue'
import { SvgIcon } from '@/components/icons/SvgIcon.tsx'
import { NumberOutput } from '@/modules/items/NumberOutput.tsx'
import { Numpad } from '@/modules/items/Numpad.tsx'
import { time } from '@/shared/time.ts'
import { DatePicker, Popup } from 'vant'

export const InputPad = defineComponent({
  name: 'InputPad',
  setup(/* _props, _ctx */) {
    const datePickerVisibleRef = ref(false)
    const lastDateRef = ref(['2026', '11', '01'])
    const currentDateRef = ref(['2026', '11', '01'])
    const minDate = new Date(2025, 1, 1)
    const maxDate = new Date(2028, 0, 1)
    const formattedDateStr = computed(() =>
      time(new Date(currentDateRef.value.join('-'))).api.format(),
    )
    const toggleDatePicker = () => {
      datePickerVisibleRef.value = !datePickerVisibleRef.value
    }
    const confirmDate = ({ selectedValues }: {selectedValues: string[]}) => {
      lastDateRef.value = selectedValues
      currentDateRef.value = selectedValues
      datePickerVisibleRef.value = false
    }
    const cancelDate = () => {
      currentDateRef.value = lastDateRef.value
      datePickerVisibleRef.value = false
    }

    return () => (
      <>
        {/* 日期、金额输入 */}
        <div class="flex justify-between items-center py-1em b-t-[var(--button-border-color)] b-t-1">
          <span class="flex justify-start items-center">
            <SvgIcon name="date" class="size-2em mx-1 fill-[var(--button-bg)]" />
            <span class="animation-rainbow text-14px font-mono color-[var(--vp-c-brand-next)]">
              <span onClick={toggleDatePicker}>{formattedDateStr.value}</span>
              <Popup
                position="bottom"
                v-model:show={datePickerVisibleRef.value}
                onClose={cancelDate}>
                <DatePicker
                  title="选择年月日"
                  v-model={currentDateRef.value}
                  minDate={minDate}
                  maxDate={maxDate}
                  onConfirm={confirmDate}
                  onCancel={cancelDate}/>
              </Popup>
            </span>
          </span>

          <NumberOutput class="mr-4" />
        </div>

        {/* 数字键盘 */}
        <Numpad />
      </>
    )
  },
})
  • datePickerVisibleRef 控制 Popup 显示状态
  • lastDateRef 为 DatePicker 暂存上一次选择数据，用来取消时，恢复上一次数据
  • currentDateRef 为 DatePicker 当前选择数据
  • DatePicker 设置范围 minDate、maxDate，格式为 new Date(2028, 0, 1)
  • formattedDateStr 配合自己封装的 time 函数，计算格式化后的字符串
  • toggleDatePicker 切换开/关 Popup
  • confirmDate 确认选择日期，同时更新 lastDateRef 和 currentDateRef
  • cancelDate 取消选择日期，由于切换日期时组件立即就会同时更新 currentDateRef 数据，取消时需要恢复上一次选择的数据 lastDateRef

import { defineStore } from 'pinia'
import { ref } from 'vue'
import { showNotify } from 'vant'

export const useNumpadAmountStore = defineStore('numpadAmount', () => {
  const amountRef = ref('0.00')

  const appendText = (inputText: string) => {
    // 如果当前金额为'0.00'，则清空金额
    if(amountRef.value === '0.00') { amountRef.value = ''}
    // 小数点后已经有两位数字，则不能再输入
    if(amountRef.value.match(/\.\d{2}/)) {inputText = ''}
    // 已经有'.'，则不能再输入'.'
    if(amountRef.value.match(/\./) && inputText === '.') {inputText = ''}
    // 输入'.'，判断当前金额是否为空，是则首位补零
    if(inputText === '.' && amountRef.value === '') {
      amountRef.value = '0'
    }
    // 输入 '0'，判断金额是否为'0'，是则不输入
    if(amountRef.value === '0' && inputText === '0') {
      inputText = ''
    }
    // 输入 '1' ~' 9'.判断金额是否为'0'，是则替换为1~9
    if((/[1-9]/g).test(inputText) && amountRef.value === '0') {
      amountRef.value = inputText
      inputText = ''
    }
    // 位数超过10位数字，则不能再输入，并提示
    if((amountRef.value).replace('.', '').length > 10) {
      showNotify({ type: 'warning', position: 'bottom', message: '输入的位数不支持超过10位', })
      inputText = ''
    }
    amountRef.value += inputText
  }

  const clearText = () => { amountRef.value = '0.00' }

  const backspaceText = () => {
    amountRef.value = amountRef.value.slice(0, -1)
    // 删空则补零
    if(amountRef.value === '') { amountRef.value = '0'}
  }

  return {
    amountRef,
    appendText,
    clearText,
    backspaceText,
  }
})

  if (typeof hint === 'string') {

    showNotify(hint)
