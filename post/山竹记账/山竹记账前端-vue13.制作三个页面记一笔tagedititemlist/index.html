    import { defineComponent, type PropType } from 'vue'
import { ItemTab } from '@/components/tab/ItemTab.tsx'
import { cn } from 'cn'

export const SwitchTabs = defineComponent({
name: 'SwitchTabs',
props: {
selected: {
  type: String as PropType<string>,
},
},
setup(_props, _ctx) {
const selectedClass = (tabName: string) =>
  _props.selected === tabName
    ? cn(
        'relative',
        `after:(content-[""] block absolute bottom-0 left-0 w-full h-4px bg-[var(--tabs-indicator-bg)])`,
      )
    : ''

return () => {
  const defaultSlotTabs = _ctx.slots?.default?.() ?? []
  const ifAllItemTab = defaultSlotTabs.every(
    (slot) => slot.type === ItemTab && !!slot.props?.tabName,
  )
  if (!ifAllItemTab) {
    throw new Error(`SwitchTabs: 子组件必须是 ItemTab，且必须有 tabName 属性`)
  }

  return (
    <div class="grow-1 shrink-1 overflow-auto scrollbar-hide relative">
      {/* 切换标签 */}
      <nav class="sticky left-0 top-0 z-1">
        <menu class="flex items-center text-color-[var(--navbar-text-color)]">
          {defaultSlotTabs.map((slot, index) => (
            <li
              key={index + slot?.props?.tabName}
              class={cn(
                'grow-1 shrink-0 text-center',
                'p-y-12px bg-[var(--navbar-bg-end)]',
                selectedClass(slot?.props?.tabName),
              )}
              onClick={() => {
                _ctx.emit('update:selected', slot?.props?.tabName)
              }}>
              {slot?.props?.tabName}
            </li>
          ))}
        </menu>
      </nav>

      {/* 标签内容-插槽-数组 */}
      <section>
        {defaultSlotTabs.find((slot) => slot.props?.tabName === _props.selected)}
      </section>
    </div>
  )
}
},
})

    tagName: { type: String as PropType<string>, default: '', },
    tagSign: { type: String as PropType<string>, default: '😃', },
  },
  setup( _props, _ctx ) {
    return () => (
      <div class="relative w-20vw h-[calc(20vw*1.14)] flex flex-col items-center">

import { defineComponent, ref } from 'vue'
import { MainLayout } from '@/components/layout/MainLayout.tsx'
import { SvgIcon } from '@/components/icons/SvgIcon.tsx'
import { SwitchTabs } from '@/components/tab/SwitchTabs.tsx'
import { ItemTab } from '@/components/tab/ItemTab.tsx'

export const ItemList = defineComponent({
  name: 'ItemList',
  setup(/* _props, _ctx */) {
    const selectedDateRef = ref('本月')
    return () => (
      <MainLayout>
        {{
          title: () => '山竹记账',
          icon: () => <SvgIcon name="menu" class="w-2em h-2em" />,
          default: () => (
            <div class="h-full flex flex-col">
              <SwitchTabs
                selected={selectedDateRef.value}
                v-model:selected={selectedDateRef.value}
                class="[&>nav>menu>li]:(text-start grow-0)">
                <ItemTab tabName="本月" tabId="thisMonth">本月</ItemTab>
                <ItemTab tabName="上月" tabId="lastMonth">上月</ItemTab>
                <ItemTab tabName="今年" tabId="thisYear">今年</ItemTab>
                <ItemTab tabName="自定义范围" tabId="customRange">自定义范围</ItemTab>
              </SwitchTabs>
            </div>
          ),
        }}
      </MainLayout>
    )
  },
})
  • 使用父级样式覆盖 '[&>nav>menu>li]:(text-start grow-0)'
