import { cn } from "cn";

interface Props<T> {
  className?: string
  selectedTabKey: T
  tabItems: Readonly<Array<{ key: T; text: string }>>
  onSwitchTab: (key: T) => void
}

export const SwitchTabs = <T extends string>({ className, tabItems, selectedTabKey, onSwitchTab, }: Props<T>) => {
  const selectedClass = (key: string) => cn(
    'relative',
    key === selectedTabKey
      ? 'after:(content-[""] block absolute bottom-0 left-0 w-full h-4px bg-[var(--tabs-indicator-bg)])'
      : ''
  )

  return (
    <nav className={className}>
      <menu className={cn(
        'flex text-white',
        'children:(px-24px py-12px cursor-pointer)',
      )}>
        {
          tabItems.map(item => (
            <li
              key={item.key}
              onClick={() => onSwitchTab(item.key)}
              className={selectedClass(item.key)}>
              {item.text}
            </li>
          ))
        }
      </menu>
    </nav>
  )
}
  • 去除函数组建声明 : React.FC<Props>
  • 参数前声明泛型 <T extends string>
    • 泛型参数 T 用于限制 tabItems 中的 key 类型为 string
  • 参数后声明类型 { className, tabItems, selectedTabKey, onSwitchTab, }: Props<T>)
  • 声明外部参数 interface Props<T> 传类型参数 T
    • 声明 selectedTabKey: T
    • 声明 tabItems: Readonly<Array<{ key: T; text: string }>>
    • 声明 onSwitchTab: (key: T) => void

import { cn } from "cn";

interface Props<T> {
  className?: string
  selectedTabKey: T
  tabItems: Readonly<Array<{ key: T; text: string; tabContent?: React.ReactNode }>>
  onSwitchTab: (key: T) => void
}

export const SwitchTabs = <T extends string>({
  className,
  selectedTabKey,
  tabItems,
  onSwitchTab,
}: Props<T>) => {
  const selectedClass = (key: string) => cn(
    'relative',
    key === selectedTabKey
      ? 'after:(content-[""] block absolute bottom-0 left-0 w-full h-4px bg-[var(--tabs-indicator-bg)])'
      : ''
  )

  return (
    <div className={cn(className,)}>
      {/* 切换标签导航 */}
      <nav className={cn('tab-nav',)}>
        <menu className={cn(
          'tab-menu',
          'flex text-white',
          'children:(px-24px py-12px cursor-pointer)',
        )}>
          {
            tabItems.map(item => (
              <li
                key={item.key}
                onClick={() => onSwitchTab(item.key)}
                className={cn(
                  selectedClass(item.key),
                  'tab-menu-item',
                )}>
                {item.text}
              </li>
            ))
          }
        </menu>
      </nav>

      {/* 导航内容 */}
      <section className={cn( 'tab-content', )}>
        {tabItems.find(item => item.key === selectedTabKey)?.tabContent}
      </section>
    </div>
  )
}
  • 添加传参 tabContent?: React.ReactNode
    • tabItems: Readonly<Array<{ key: T; text: string; tabContent?: React.ReactNode }>>
  • 渲染导航内容 {tabItems.find(item => item.key === selectedTabKey)?.tabContent}

import { cn } from "cn";

interface Props<T> {
  className?: string
  classPrefix?: string
  selectedTabKey: T
  tabItems: Readonly<Array<{ key: T; text: string; tabContent?: React.ReactNode }>>
  onSwitchTab: (key: T) => void
}

export const SwitchTabs = <T extends string>({
  className,
  classPrefix,
  selectedTabKey,
  tabItems,
  onSwitchTab,
}: Props<T>) => {
  const selectedClass = (key: string) => cn(
    'relative',
    key === selectedTabKey
      ? 'after:(content-[""] block absolute bottom-0 left-0 w-full h-4px bg-[var(--tabs-indicator-bg)])'
      : ''
  )

  return (
    <div className={cn(className, classPrefix)}>
      {/* 切换标签导航 */}
      <nav className={cn(classPrefix ? `${classPrefix}-tab-nav` : 'tab-nav',)}>
        <menu className={cn(
          classPrefix ? `${classPrefix}-tab-menu` : 'tab-menu',
          'flex text-white',
          'children:(px-24px py-12px cursor-pointer)',
        )}>
          {
            tabItems.map(item => (
              <li
                key={item.key}
                onClick={() => onSwitchTab(item.key)}
                className={cn(
                  selectedClass(item.key),
                  classPrefix ? `${classPrefix}-tab-menu-item` : 'tab-menu-item',
                )}>
                {item.text}
              </li>
            ))
          }
        </menu>
      </nav>

      {/* 导航内容 */}
      <section className={cn(
        classPrefix ? `${classPrefix}-tab-content` : 'tab-content',
      )}>
        {tabItems.find(item => item.key === selectedTabKey)?.tabContent}
      </section>
    </div>
  )
}
  • 内部结构样式
    • classPrefix > ${classPrefix}-tab-nav > ${classPrefix}-tab-menu > ${classPrefix}-tab-menu-item
    • classPrefix > ${classPrefix}-tab-content

import styled from 'styled-components';

// 1. 定义 Props 接口
interface ButtonProps {
  $primary?: boolean;
  $size?: 'small' | 'large';
}

// 2. 使用泛型传参
const Button = styled.button<ButtonProps>`
  background: ${props => props.$primary ? 'blue' : 'gray'};
  padding: ${props => props.$size === 'large' ? '12px 24px' : '8px 16px'};
  color: white;
  border: none;
`

// 3. 在组件中使用
export const App = () => (
  <Button $primary $size="large">点击我</Button>
)
  • 注意上面代码中属性名前面的 $ 符号。这是 styled-components 的 Transient Props 特性
  • 带有 $ 的属性只会在样式组件中生效，不会被透传到最终的 HTML 标签上
    • 避免抛出 React does not recognize the ... prop on a DOM element 的警告

import styled from 'styled-components'

// 一个基础的泛型列表组件
interface ListProps<T> {
  items: T[];
  renderItem: (item: T) => React.ReactNode;
  className?: string; // 必须包裹 className 才能被 styled() 继承样式
}
function List<T>({ items, renderItem, className }: ListProps<T>) {
  return <div className={className}>{items.map(renderItem)}</div>
}

// 为泛型组件包裹样式
const StyledList = styled(List)`
  background-color: #f5f5f5;
  padding: 200px;
` as typeof List // 使用 as typeof 保持原组件的泛型能力
  • 在 Styled Components 组件后添加断言 as typeof 保持原组件的泛型能力
  • 必须包裹 className 才能被 styled() 继承样式
