1. v-model 的正确用法 ⇧

  • 双向绑定数据
    • 父组件中在子组件上 v-model={} 即绑定默认的值 v-model:modelValue={}
    • 相当于写了 modelValue={} onUpdate:modelValue={() => {}} 的语法糖
      • 写了 onUpdate:modelValue 子组件内必须写 emits: ['update:modelValue',]
    • 在 input 上使用 v-model={} onInput={() => {}}
    • 绑自定义数据 v-model:customValue={} 注意只用小驼峰命名绑定值
    • 不可用下划线等其他形式，特别的下划线用来指定原来 SFC 中修饰符
    • 子组件 defineComponent 选项中写
    • props: { modelValue: {}, customValue: {} },
    • emits: ['update:modelValue', 'update:customValue']

import { defineComponent, type PropType, } from 'vue'
import { ItemTab } from '@/components/tab/ItemTab.tsx'
import { cn } from 'cn'
import { genDefaultSlotTabs } from '@/views/genDefaultSlotTabs.ts'

export const SwitchTabs = defineComponent({
  name: 'SwitchTabs',
  props: {
    selected: {
      type: String as PropType<string>,
    },
  },
  emits: [
    // 由点击事件触发
    'update:selected',
  ],
  setup(_props, _ctx) {
    const selectedClass = (tabName: string) =>
      _props.selected === tabName
        ? cn(
            'relative',
            `after:(content-[""] block absolute bottom-0 left-0 w-full h-4px bg-[var(--tabs-indicator-bg)])`,
          )
        : ''
    
    const onSwitchTab = (tabName: string) => {
      _ctx.emit('update:selected', tabName)
    }

    return () => {
      const defaultSlotTabs = genDefaultSlotTabs(_ctx.slots?.default?.() ?? [])
      const ifAllItemTab = defaultSlotTabs
        .every((slot) => slot.type === ItemTab && !!slot.props?.tabName)
      if (!ifAllItemTab) {
        throw new Error(`SwitchTabs: 子组件必须是 ItemTab，且必须有 tabName 属性`)
      }

      return (
        <div class="grow-1 shrink-1 overflow-auto scrollbar-hide relative">
          {/* 切换标签 */}
          <nav class="sticky left-0 top-0 z-1 bg-[var(--navbar-bg-end)]">
            <menu class="flex justify-start items-center text-color-[var(--navbar-text-color)]">
              {defaultSlotTabs.map((slot, index) => (
                <li
                  key={index + slot?.props?.tabName}
                  class={cn(
                    'where:(text-center grow-1 shrink-0 px-1em py-12px)',
                    selectedClass(slot?.props?.tabName),
                  )}
                  onClick={() => onSwitchTab(slot?.props?.tabName)}>
                  {slot?.props?.tabName}
                </li>
              ))}
            </menu>
          </nav>

          {/* 标签内容-插槽-数组 */}
          <section class="max-h-[calc(100vh-136px)] overflow-auto scrollbar-hide">
            {defaultSlotTabs.find((slot) => slot.props?.tabName === _props.selected)}
          </section>
        </div>
      )
    }
  },
})
  • 必须添加
    • emits: ['update:selected',]
    • Props: { selected: {}, }

export const ItemList = defineComponent({
  name: 'ItemList',
  setup(/* _props, _ctx */) {
    const selectedDateRef = ref('本月')
    const currentDate = new DateTime()
    const customDate = reactive<{ start: string; end: string }>({
      start: currentDate.getOriginDate()?.format(),
      end: currentDate.getOriginDate()?.format()
    })

    const dateList = [
      {start: currentDate.firstDayOfMonth().format(), end: currentDate.lastDayOfMonth().format(),},
      {
        start: currentDate.subtract(1, 'month').firstDayOfMonth().format(),
        end: currentDate.subtract(1, 'month').lastDayOfMonth().format(),
      },
      {start: currentDate.firstDayOfYear().format(), end: currentDate.lastDayOfYear().format(),},
    ] as const

    const customDateRangeOverlayVisibleRef = ref(false)
    const onSubmitCustomDateRange = (e: SubmitEvent) => {
      e.preventDefault()
      customDateRangeOverlayVisibleRef.value = false
    }
    const onCancelCustomDateRange = () => {
      customDateRangeOverlayVisibleRef.value = false
    }
    const openCustomDateRange = () => {
      customDateRangeOverlayVisibleRef.value = selectedDateRef.value === '自定义日期范围'
    }

    return () => (
      <MainLayout>
        {{
          title: () => '山竹记账',
          icon: () => <SvgIcon name="menu" class="w-2em h-2em" />,
          default: () => (
            <div class="h-full flex flex-col">
              <SwitchTabs
                selected={selectedDateRef.value}
                v-model:selected={selectedDateRef.value}
                onUpdate:selected={openCustomDateRange}
                class="[&>nav>menu>li]:(text-start grow-0)">
                <ItemTab tabName="本月" tabId="thisMonth">
                  <ItemSummary
                    startDate={dateList[0].start}
                    endDate={dateList[0].end} />
                </ItemTab>
                <ItemTab tabName="上月" tabId="lastMonth">
                  <ItemSummary
                    startDate={dateList[1].start}
                    endDate={dateList[1].end} />
                </ItemTab>
                <ItemTab tabName="今年" tabId="thisYear">
                  <ItemSummary
                    startDate={dateList[2].start}
                    endDate={dateList[2].end} />
                </ItemTab>
                <ItemTab tabName="自定义日期范围" tabId="customDateRange">
                  <ItemSummary
                    startDate={customDate.start}
                    endDate={customDate.end} />
                </ItemTab>
              </SwitchTabs>

              {/* 自定义日期范围弹出层 */}
              <Overlay
                show={customDateRangeOverlayVisibleRef.value}
                onClick={onCancelCustomDateRange}
                class="flex items-center justify-center bg-[var(--overlay-mask-bg)]" >
                <div
                  onClick={(e: MouseEvent) => e.stopPropagation()}
                  class="bg-[var(--overlay-bg)] w-80vw h-35vh">
                  <header class="bg-[var(--main)] text-[var(--dialog-header-text)] py-12px px-1em">
                    请选择时间
                  </header>
                  <main>
                    <FormContainer onSubmit={onSubmitCustomDateRange}>
                      <FormItem type="date" label="开始时间" v-model={customDate.start} />
                      <FormItem type="date" label="结束时间" v-model={customDate.end} />
                      <footer class="flex justify-between mt-1em children:(w-[calc(50%-.25em)])">
                        <FormItem type="button" onClick={onCancelCustomDateRange} buttonConfig={{level: 'default', text: '取 消' }}/>
                        <FormItem type="submit-button" buttonConfig={{level: 'primary', text: '确 认' }}/>
                      </footer>
                    </FormContainer>
                  </main>
                </div>
              </Overlay>

              <FloatButton class="fixed bottom-1em right-1em" iconName="add" />
            </div>
          ),
        }}
      </MainLayout>
    )
  },
})

import { computed, defineComponent, type PropType, } from 'vue'
import { cn } from 'cn'

export const InputText = defineComponent({
  name: 'InputText',
  props: {
    error: { type: String as PropType<string>, },
    forId: { type: String as PropType<string>, },
    modelValue: { type: String as PropType<string>, default: '', },
  },
  setup(_props, _ctx) {
    const getModelValue = computed({
      get() {
        return _props.modelValue?.toString() ?? ''
      },
      set(value) {
        _ctx.emit('update:modelValue', value)
      },
    })
    return () => (
      <input
        id={_props.forId}
        v-model={getModelValue.value}
        class={cn(
          'p-x-1em text-18px',
          'b-1 rounded-[var(--input-radius)] shadow-[inset_0_0_3px_var(--input-shadow)]',
          _props.error ? 'b-[var(--error-color)]' : 'b-[var(--input-border-color)]',
        )}
      />
    )
  },
})

import { computed, defineComponent, type PropType, ref, useId } from 'vue'
import { EmojiSelector } from '@/modules/tags/EmojiSelector.tsx'
import type { FormDataType } from '@/shared/validator/formValidate.ts'
import { BarButton, type ButtonLevelType } from '@/components/BarButton.tsx'
import { LabelOrDivWrapper } from '@/components/form/items/LabelOrDivWrapper.tsx'
import { InputText } from '@/components/form/items/InputText.tsx'
import { InputDatePickerPopup } from '@/components/form/items/InputDatePickerPopup.tsx'
import { FormItemContentSlot } from '@/components/form/items/FormItemContentSlot.tsx'

export type FormItemType = 'text' | 'emojiSelector' | 'date' | 'submit' | 'button'

interface FormItemProps {
  type?: FormItemType
  modelValue?: FormDataType[string]
  visibleValue?: boolean
  error?: string
  hasErrorPlaceholder?: boolean
  label?: string
  hasLabelColon?: boolean
  buttonConfig?: { level?: ButtonLevelType, text?: string }
  onClick?: () => void
}

export const FormItem = defineComponent<FormItemProps>({
  name: 'FormItem',
  props: {
    type: { type: String as PropType<FormItemType>, },
    label: { type: String as PropType<string>, default: '', },
    hasLabelColon: { type: Boolean as PropType<boolean>, default: true, },
    modelValue: {
      type: [String, Number, null, undefined,] as PropType<FormDataType[string]>,
      default: '',
    },
    visibleValue: { type: Boolean as PropType<boolean>, default: false, },
    error: { type: String as PropType<string>, },
    hasErrorPlaceholder: { type: Boolean as PropType<boolean>, default: false, },
    buttonConfig: {
      type: Object as PropType<{ level?: ButtonLevelType, text?: string }>,
      default: () => ({ level: 'default', text: '确认' }),
    },
  },
  emit: ['update:modelValue', 'update:visibleValue',],
  setup( _props, _ctx ) {
    const extraHint = ref('')
    const getModelValue = computed({
      get() {return _props.modelValue?.toString()},
      set(value) {_ctx.emit('update:modelValue', value)}
    })
    const getVisibleValue = computed({
      get() {return _props.visibleValue},
      set(value) {_ctx.emit('update:visibleValue', value)}
    })
    const getIfNeedLabelFor = computed(() =>
      ['text', 'submit', 'button', 'date'].includes(_props?.type ?? ''))
    const forId = useId()
    const getLabelFor = computed(() => getIfNeedLabelFor.value
        ? `form-item-${_props.type}-${forId}`
        : '')
    const itemTypeSwitchContent = computed(() => {
      switch (_props.type) {
        case 'text':
          return (
            <InputText
              error={_props.error}
              v-model={getModelValue.value}
              forId={getLabelFor.value} />
          )
        case 'emojiSelector':
          return (
            <EmojiSelector
              v-model={getModelValue.value}
              v-model:extraHint={extraHint.value}
              hasError={!!_props.error} />
          )
        case 'submit':
        case 'button':
          return <BarButton
                  level={_props.buttonConfig?.level}
                  forId={getLabelFor.value}
                  type={_props.type}
                  text={_props.buttonConfig?.text}/>
        case 'date':
          return (
            <InputDatePickerPopup
              forId={getLabelFor.value}
              v-model={getModelValue.value}
              v-model:visibleValue={getVisibleValue.value}
              error={_props.error}/>
          )
        case undefined:
          return _ctx.slots.default?.()
      }
    })
    return () => (
      <div class="mt-.5em">
        <LabelOrDivWrapper
          ifUseLabelTag={getIfNeedLabelFor.value}
          labelFor={getLabelFor.value}>
          <FormItemContentSlot
            label={_props.label}
            hasLabelColon={_props.hasLabelColon}
            error={_props.error}
            hasErrorPlaceholder={_props.hasErrorPlaceholder}
            extraHint={extraHint.value}>
            {itemTypeSwitchContent.value}
          </FormItemContentSlot>
        </LabelOrDivWrapper>
      </div>
    )
  },
})
  • 表单项类型目前封装有 InputText EmojiSelector BarButton InputDatePickerPopup，方便扩展
  • 内容展示插槽组件 FormItemContentSlot 传插槽 {itemTypeSwitchContent.value}
  • useId 生成的每个 ID 在每个应用内都是唯一的
    • 不应在 computed() 属性内部调用，因为这可能导致实例冲突
    • 应在 computed() 外部声明 ID，并在计算函数内部引用它
