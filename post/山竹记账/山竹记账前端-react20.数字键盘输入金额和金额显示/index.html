默认显示 ¥ 0.00

  • 输入有效数字和小数点
    • 初始输入 1-9，显示 1-9
    • 之后输入任意数字，显示对应数字追加到后一位，但不超过最长允许长度
    • 之后输入 .，显示整数部分 + .
      • 之后输入任意数字，小数部分只可输入最多两位数字
      • 之后再输入 .，不往后追加，仍显示原来
    • 初始输入 0，显示 0
    • 之后输入任意数字，去除首位的零，替换为输入的数字
    • 之后输入 .，显示 0.
      • 之后输入任意数字，小数部分只可输入最多两位数字
      • 之后再输入 .，不往后追加，仍显示原来
    • 初始输入 .，显示 0.
    • 之后输入任意数字，小数部分只可输入最多两位数字
    • 之后再输入 .，不往后追加，仍显示原来
  • 点击删除按钮
    • 一次只能删除最后一位数字或 .
    • 删除到只剩最后一位数字，再删除，恢复为初始默认显示 ¥ 0.00
  • 点击归零按钮，都恢复为初始默认显示 ¥ 0.00

import { create } from 'zustand'

interface NumpadAmountState {
  amount: string
}
interface NumpadAmountActions {
  appendText: (inputText: string) => void
  backspaceText: () => void
  clearText: () => void
}

export const useNumpadAmountStore = create<NumpadAmountState & NumpadAmountActions>()(
  (set, get) => ({
    amount: '0.00',
    appendText: (inputText: string) => {
      const conditions = [
        {
          ruleHint: `输入时，如果当前金额为'0.00'，则清空金额`,
          check: () => get().amount === '0.00',
          action: () => {
            set({ amount: '' })
          },
        },
        {
          ruleHint: `小数点后已经有两位数字，则不能再输入`,
          check: () => /\.\d{2}/.test(get().amount),
          action: () => {
            inputText = ''
          },
        },
        {
          ruleHint: `已经有'.'，则不能再输入'.'`,
          check: () => get().amount.includes('.') && inputText === '.',
          action: () => {
            inputText = ''
          },
        },
        {
          ruleHint: `输入'.'，判断当前金额是否为空，是则首位补零`,
          check: () => inputText === '.' && get().amount === '',
          action: () => {
            set({ amount: '0' })
          },
        },
        {
          ruleHint: `输入 '0'，判断金额是否为'0'，是则不输入`,
          check: () => get().amount === '0' && inputText === '0',
          action: () => {
            inputText = ''
          },
        },
        {
          ruleHint: `输入 '1' ~ ' 9'，判断当前金额是否为'0'，是则替换为 '1' ~ ' 9'`,
          check: () => /[1-9]/g.test(inputText) && get().amount === '0',
          action: () => {
            set({ amount: inputText })
            inputText = ''
          },
        },
        {
          ruleHint: `位数超过10位数字，则不能再输入，并提示`,
          check: () => get().amount.replace('.', '').length >= 10,
          action: () => {
            inputText = ''
            console.warn('输入的数字长度不支持超过10位')
            // TODO: 待使用事件总线解耦通知逻辑
          },
        },
      ]

      conditions?.forEach((condition) => {
        if (condition.check()) {
          condition.action()
        }
      })

      set((state) => ({ amount: state.amount + inputText }))
    },
    backspaceText: () => {
      set((state) => ({ amount: state.amount.slice(0, -1) }))
      // 删空则补零
      if (get().amount === '') {
        set({ amount: '0.00' })
      }
    },
    clearText: () => {
      set({ amount: '0.00' })
    },
  })
)
  • 一个数据状态 amount 读写金额状态
  • 三个逻辑：
    • appendText
    • 表驱动 遍历条件修改：
      • 输入参数 inputText
      • 显示金额 amount
      • 拼接逻辑 set((state) => ({ amount: state.amount + inputText }))
    • backspaceText
    • clearText
  • 重构替换 useState 拦截器写法

import { createNanoEvents } from 'nanoevents'
import { SweetAlertOptions } from 'sweetalert2'

type HintType = 'warning' | 'error' | 'success' | 'info' | 'question'
type HintPosition =
  | 'top'
  | 'top-start'
  | 'top-end'
  | 'center'
  | 'center-start'
  | 'center-end'
  | 'bottom'
  | 'bottom-start'
  | 'bottom-end'

export type HintParams = {
  type: HintType
  message: string
  position: HintPosition
} & SweetAlertOptions

export type HintPayload = undefined | string | HintParams

export interface GlobalHintsEvents {
  hint: (payload?: HintPayload) => void
  // noPayloadEventName: () => void
}

export const emitter = createNanoEvents<GlobalHintsEvents>()

/**
 * 创建总线单例
 * 导出两个接口，可直接解构获取
 * 1. on: 监听总线事件，返回一个取消监听函数
 * 2. emit: 触发总线事件
 */
export const useGlobalHintsBus = () => ({
  ...(emitter ? emitter : createNanoEvents<GlobalHintsEvents>()),
  on: emitter.on.bind(emitter),
  emit: emitter.emit.bind(emitter),
})
  • 自定义提示事件类型写在 GlobalHintsEvents 中
  • 配合使用 sweetalert2 提示框，payload 传参类型参考 SweetAlertOptions

export const throttledFn = <F extends (...args: any[]) => unknown>(fn: F, delay: number = 500) => {
  let timer: NodeJS.Timeout | null = null
  let result: ReturnType<F>
  return (...args: Parameters<F>) => {
    if (timer) {
      return result
    }
    result = fn(...args) as ReturnType<F>
    timer = setTimeout(() => {
      clearTimeout(timer as NodeJS.Timeout)
      timer = null
    }, delay)
    return result
  }
}

function triggerHapticFeedback(ms = 15) {
    if ('vibrate' in navigator) {
        // 参数为震动时长（毫秒）
        // 按钮点击通常使用 15ms-20ms 即可，太长会显得沉重、黏糊
        navigator.vibrate(ms)
    }
}

// 绑定到按钮的 click 或 touchstart 事件上
const btn = document.querySelector('.vibrate-btn')
btn.addEventListener('click', () => {
    triggerHapticFeedback(15)
})

    import { useEffect, useRef, useCallback } from 'react'

type RawTriggerMap = Record<string, (...args: any[]) => any>

export interface UseVibrateOptions {
/**
* 是否循环震动
* @default false
*/
isLoop?: boolean
rawTriggerMap?: RawTriggerMap
}

export const useVibrate = (
pattern: VibratePattern = 15, // 默认轻触反馈 15ms
options: UseVibrateOptions = {}
) => {
const { isLoop = false, rawTriggerMap } = options
const intervalRef = useRef<number | null>(null)

// 检查当前浏览器是否支持
const isSupported = typeof navigator !== 'undefined' && 'vibrate' in navigator

// 1. 核心震动方法
const trigger = useCallback(
(customPattern?: VibratePattern) => {
  if (!isSupported) return
  navigator.vibrate(customPattern ?? pattern)
},
[isSupported, pattern]
)

// 2. 停止震动方法
const stop = useCallback(() => {
if (!isSupported) return
if (intervalRef.current) {
  clearInterval(intervalRef.current)
  intervalRef.current = null
}
if (!navigator.userActivation.hasBeenActive) return
navigator?.vibrate?.(0) // 传入 0 即可立即停止当前所有震动
}, [isSupported])

// 3. 启动循环震动
const start = useCallback(() => {
if (!isSupported) return
stop() // 启动前先清除上一次的定时器

// 如果是数组，计算一个完整的震动周期总耗时
const duration = Array.isArray(pattern)
  ? pattern.reduce((acc, cur) => acc + cur, 0)
  : (pattern as number)

trigger() // 先执行第一次

if (isLoop) {
  intervalRef.current = window.setInterval(() => {
    trigger()
  }, duration + 50) // 加 50ms 缓冲，避免硬件响应不过来导致连续震动粘连
}
}, [isSupported, pattern, isLoop, trigger, stop])

// 组件卸载时自动清理，防止内存泄漏或后台持续震动
useEffect(() => {
return () => stop()
}, [stop])

// 为每一个rawTriggerMap的key创建一个震动触发器，然后映射到triggerMap中
const triggerMap = rawTriggerMap
? Object.keys(rawTriggerMap).reduce((acc, key) => {
    acc[key] = (...args) => {
      trigger()
      rawTriggerMap[key](...args)
    }
    return acc
  }, {} as RawTriggerMap)
: {}

return {
isSupported,
vibrate: trigger, // 触发单次/传入模式震动
start, // 开始循环（如果配置了 isLoop: true）
stop, // 停止震动
triggerMap, // 创建的触发器
}
}
