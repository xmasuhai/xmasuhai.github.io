export class Time {
  #date: Date
  constructor(date = new Date()) {
    this.#date = date
  }
  format(pattern: string = 'YYYY-MM-DD') {
    // YYYY
    const year = this.#date.getFullYear()
    // MM
    const month = this.#date.getMonth() + 1
    // DD
    const day = this.#date.getDate()
    // HH
    const hour = this.#date.getHours()
    // mm
    const minute = this.#date.getMinutes()
    // ss
    const second = this.#date.getSeconds()
    // sss
    const millisecond = this.#date.getMilliseconds()
    return (
      pattern
        .replace(/YYYY/g, year.toString())
        // 使用 padStart 长度不足则补零
        .replace(/MM/g, month.toString().padStart(2, '0'))
        .replace(/DD/g, day.toString().padStart(2, '0'))
        .replace(/HH/g, hour.toString().padStart(2, '0'))
        .replace(/mm/g, minute.toString().padStart(2, '0'))
        .replace(/ss/g, second.toString().padStart(2, '0'))
        .replace(/sss/g, millisecond.toString().padStart(3, '0'))
    )
  }
}

add(amount: number, unit: 'year' | 'month' | 'week' | 'day' | 'hour' | 'minute' | 'second' | 'millisecond') {
  const date = new Date(this.#date.getTime())
  switch (unit) {
    case 'year':
      date.setFullYear(date.getFullYear() + amount)
      break
    case 'month':
      date.setMonth(date.getMonth() + amount)
      break
    case 'week':
      date.setDate(date.getDate() + amount * 7)
      break
    case 'day':
      date.setDate(date.getDate() + amount)
      break
    default:
      throw new Error(`Invalid unit: ${unit}`)
  }
  • 当前月份第一天的日期
    • 使用 this.date.setDate(1) 会改当前调用的日期，不符合预期，并不想修改日期，而是创建一个新的
    • 改用 new Time(new Date()) 来构造：new Time(new Date(this.date.getFullYear(), this.date.getMonth(), 1, 0,0,0))
    • 当前月的最后一天就是下月的0号
  • 暂不考虑闰秒的情况
  • add 需要考虑月份的特殊情况 date.setMonth(date.getMonth() + amount)
    • const d1 = new Date(2000, 0, 31) 闰年1月份加上一个月的日期
    • d1 为 Mon Jan 31 2000 00:00:00 GMT+0800 (中国标准时间) 符合预期
    • d1.setMonth(d1.getMonth() + 1)，再查看
    • d1 变为 Thu Mar 02 2000 00:00:00 GMT+0800 (中国标准时间) 不符合预期
    • 加上的是当前月份的天数：一月加一个月就是加上31天，二月加一个月就是加上28或者29天
    • 1月31号加29天应该是29号，因为2000年是闰年
    • 需要改为人类容易理解的判断方式

export class DateTime {
  #date: Date
  constructor(date = new Date()) {
    this.#date = date
  }
  format(pattern: string = 'YYYY-MM-DD') {
    // YYYY
    const year = this.#date.getFullYear()
    // MM
    const month = this.#date.getMonth() + 1
    // DD
    const day = this.#date.getDate()
    // HH
    const hour = this.#date.getHours()
    // mm
    const minute = this.#date.getMinutes()
    // ss
    const second = this.#date.getSeconds()
    // sss
    const millisecond = this.#date.getMilliseconds()
    return (
      pattern
        .replace(/YYYY/g, year.toString())
        // 使用 padStart 长度不足则补零
        .replace(/MM/g, month.toString().padStart(2, '0'))
        .replace(/DD/g, day.toString().padStart(2, '0'))
        .replace(/HH/g, hour.toString().padStart(2, '0'))
        .replace(/mm/g, minute.toString().padStart(2, '0'))
        .replace(/ss/g, second.toString().padStart(2, '0'))
        .replace(/sss/g, millisecond.toString().padStart(3, '0'))
    )
  }
  add(amount: number, unit: 'year' | 'month' | 'week' | 'day' | 'hour' | 'minute' | 'second' | 'millisecond') {
    const date = new Date(this.#date.getTime())
    switch (unit) {
      case 'year':
        date.setFullYear(date.getFullYear() + amount)
        break
      case 'month':
        // date.setMonth(date.getMonth() + amount)
        // 01-31 + 1m = 02-28/29
        const currentDay = date.getDate() // 2000-01-31
        date.setDate(1) // 设置为 01-01，再加31天就不会跳到3月份去了，保证月份正确
        date.setMonth(date.getMonth() + amount) // 02-01
        // 获取 2000-03-00 00:00:00 的日期数字，其中 monthIndex 为 2
        const lastDayOfMonth = new Date(date.getFullYear(), date.getMonth() + 1, 0,0,0,0,0).getDate()
        date.setDate(Math.min(currentDay, lastDayOfMonth))
        break
      case 'week':
        date.setDate(date.getDate() + amount * 7)
        break
      case 'day':
        date.setDate(date.getDate() + amount)
        break
      case 'hour':
        date.setHours(date.getHours() + amount)
        break
      case 'minute':
        date.setMinutes(date.getMinutes() + amount)
        break
      case 'second':
        date.setSeconds(date.getSeconds() + amount)
        break
      case 'millisecond':
        date.setMilliseconds(date.getMilliseconds() + amount)
        break
      default:
        throw new Error(`Invalid unit: ${unit}`)
    }
  }
}

      type: [String, Number, null, undefined,] as PropType<UnwrapRef<FormDataType[string]>>,
      default: '',
    },
    error: { type: String as PropType<string>, },
    hasErrorPlaceholder: { type: Boolean as PropType<boolean>, default: false, },
  },
  emit: ['update:modelValue',],
  setup( _props, _ctx ) {
    const extraHint = ref('')
    const itemContentCommonClass = cn('grow-1 min-h-[var(--input-min-height)] max-w-100%')
    const inputClass = cn(
      'p-x-1em text-18px',
      'b-1 rounded-[var(--input-radius)] shadow-[inset_0_0_3px_var(--input-shadow)]',
      _props.error ? 'b-[var(--error-color)]' : 'b-[var(--input-border-color)]',
    )
    const getModelValue = computed({
      get() {return _props.modelValue?.toString()},
      set(value) {_ctx.emit('update:modelValue', value)}
    })
    const itemContent = computed(() => {
      switch (_props.type) {
        case 'text':
          return (
            <input
              id={`form-item-${_props.type}-${_props.uniqId}`}
              v-model={getModelValue.value}
              class={inputClass} />
          )
        case 'emojiSelector':
          return (
            <EmojiSelector
              v-model={getModelValue.value}
              hasError={!!_props.error}
              v-model:extraHint={extraHint.value} />
          )
        case 'date':
          return <input />
        case undefined:
          return _ctx.slots.default?.()
      }
    })
    return () => (
      <div class="mt-.5em">
        <label for={`form-item-${_props.type}-${_props.uniqId}`} class="block">
          {/* 标题 */}
          {_props.label &&
            <span class="label_name">
              {_props.label}{_props.hasLabelColon ? '：' : ''}{extraHint.value}
            </span>
          }
          {/* 内容 */}
          <div class={cn(itemContentCommonClass, 'content-value flex mt-0.25em children:grow-1')}>
            {itemContent.value}
          </div>
          {/* 错误 */}
          {
            (_props.error || _props.hasErrorPlaceholder)  &&
            <div class={cn('error-hint min-h-18px m-t-.25em text-12px color-[var(--error-color)]')}>
              <span>{_props.error}</span>
            </div>
          }
        </label>
      </div>
    )
  },
})
  • 父组件可使用 useId 生成唯一 id，传参 uniqId，可以区分类型相同的表单项，并且在 label 标签上的 for 属性上
    • 但不可放在 key 属性上，否则每次输入时都会重新渲染组件
  • 使用计算属性作为 v-model 的双向绑定值 const getModelValue = computed({ get() {...}, set(value) {...}, })
  • 在 itemContent 后续可继续扩展其他类型的表单项，例如 date
  • hasErrorPlaceholder 用来给存在错误提示的表单项作为占位，防止高度抖动

export const DatePickerPopup = defineComponent({
  name: 'DatePickerPopup',
  props: {
    popupVisible: { type: Boolean as PropType<boolean>, default: false, },
    dateStr: { type: String as PropType<string>, default: () => genDateStr(), }
  },
  emits: ['update:popupVisible', 'update:dateStr',],
  setup( _props, _ctx ) {
    const lastDateStrRef = ref<string>(_props.dateStr ?? genDateStr())
    const [minDate, maxDate,] = [
      new DateTime(new Date()).add(-1, 'year').getRawDate(),
      new DateTime(new Date()).add(1, 'year').getRawDate(),

      get: () => _props.dateStr.split('-'),
      // [2026, 10, 22] -> '2026-10-22'

    })
    const confirmDate = ({ selectedValues }: { selectedValues: string[] }) => {
      lastDateStrRef.value = selectedValues.join('-') // '2026-10-22'

      getPopupVisible.value = false

      getDateStrArrayTransfer.value = lastDateStrRef.value.split('-') // [2026, 10, 22]
      getPopupVisible.value = false

      <Popup
        position="bottom"
        v-model:show={getPopupVisible.value}
        onClose={cancelDate}>
        <DatePicker
          title="选择年月日"
          v-model={getDateStrArrayTransfer.value}
          minDate={minDate}
          maxDate={maxDate}
          onConfirm={confirmDate}
          onCancel={cancelDate}/>
      </Popup>
    )
  },
})

  hasLabelColon?: boolean
}

export const FormItem = defineComponent<FormItemProps>({
  name: 'FormItem',
  props: {
    label: { type: String as PropType<string>, default: '', },
    hasLabelColon: { type: Boolean as PropType<boolean>, default: true, },
    type: { type: String as PropType<FormItemType>, },
    modelValue: {
      type: [String, Number, null, undefined,] as PropType<UnwrapRef<FormDataType[string]>>,
      default: '',
    },
    error: { type: String as PropType<string>, },
    hasErrorPlaceholder: { type: Boolean as PropType<boolean>, default: false, },
  },
  emit: ['update:modelValue',],
  setup( _props, _ctx ) {
    const extraHint = ref('')
    const getModelValue = computed({
      get() {return _props.modelValue?.toString()},
      set(value) {_ctx.emit('update:modelValue', value)}
    })
    const getIfNeedLabelFor = computed(() =>
      ['text', 'submit-button'].includes(_props?.type ?? ''))
    const getLabelFor = computed(() => getIfNeedLabelFor.value
        ? `form-item-${_props.type}-${useId()}`
        : '')

    const dateVisibleRef = ref(false)
    const itemContent = computed(() => {
      switch (_props.type) {
        case 'text':
          return (
            <input
              id={getLabelFor.value}
              type="date"
              v-model={getModelValue.value}
              class={
                cn(
                  'p-x-1em text-18px',
                  'b-1 rounded-[var(--input-radius)] shadow-[inset_0_0_3px_var(--input-shadow)]',
                  _props.error ? 'b-[var(--error-color)]' : 'b-[var(--input-border-color)]',
                )
              }/>
          )
        case 'emojiSelector':
          return (
            <EmojiSelector
              v-model={getModelValue.value}
              hasError={!!_props.error}
              v-model:extraHint={extraHint.value} />
          )
        case 'submit-button':
          return <BarButton id={getLabelFor.value} type="submit">确定</BarButton>
        case 'date':
          return (
            <>
              <input
                id={getLabelFor.value}
                readonly={true}
                value={getModelValue.value}
                onClick={() => {dateVisibleRef.value = true}}
                class={
                  cn(
                    'p-x-1em text-18px',
                    'b-1 rounded-[var(--input-radius)] shadow-[inset_0_0_3px_var(--input-shadow)]',
                    _props.error ? 'b-[var(--error-color)]' : 'b-[var(--input-border-color)]',
                  )
                }/>
              <DatePickerPopup
                v-model:popupVisible={dateVisibleRef.value}
                v-model:dateStr={getModelValue.value} />
            </>
          )
        case undefined:
          return _ctx.slots.default?.()
      }
    })
    return () => (
      <div class="mt-.5em">
        <LabelOrDivWrapper
          ifUseLabelTag={getIfNeedLabelFor.value}
          labelFor={getLabelFor.value}>
          {/* 标题 */}
          {_props.label &&
            <span class="label_name">
              {_props.label}{_props.hasLabelColon ? '：' : ''}{extraHint.value}
            </span>
          }
          {/* 内容 */}
          <div class={cn('content-value flex mt-0.25em grow-1 children:grow-1 min-h-[var(--input-min-height)] max-w-100%')}>
            {itemContent.value}
          </div>
          {/* 错误 */}
          {
            (_props.error || _props.hasErrorPlaceholder)  &&
            <div class={cn('error-hint min-h-18px m-t-.25em text-12px color-[var(--error-color)]')}>
              <span>{_props.error}</span>
            </div>
          }
        </LabelOrDivWrapper>
      </div>
    )
  },
})
  • 添加 type 类型的 FormItem
    • 只读属性 readonly={true}
    • 数据为日期字符串 value={getModelValue.value}
    • 点击打开日期选择弹出框 onClick={() => {dateVisibleRef.value = true}}

import { defineComponent, reactive, ref, watchEffect } from 'vue'
import { MainLayout } from '@/components/layout/MainLayout.tsx'
import { SvgIcon } from '@/components/icons/SvgIcon.tsx'
import { SwitchTabs } from '@/components/tab/SwitchTabs.tsx'
import { ItemTab } from '@/components/tab/ItemTab.tsx'
import { ItemSummary } from '@/modules/items/ItemSummary.tsx'
import { DateTime } from '@/shared/dateTime.ts'
import { FloatButton } from '@/components/FloatButton.tsx'
import { Overlay } from 'vant'
import { FormContainer } from '@/components/form/FormContainer.tsx'
import { FormItem } from '@/components/form/FormItem.tsx'

export const ItemList = defineComponent({
  name: 'ItemList',
  setup(/* _props, _ctx */) {
    const selectedDateRef = ref('本月')
    const currentDate = new DateTime()
    const customDate = reactive<{ start: string; end: string }>({
      start: currentDate.getOriginDate()?.format(),
      end: currentDate.getOriginDate()?.format()
    })
    const dateList = [
      {start: currentDate.firstDayOfMonth(), end: currentDate.lastDayOfMonth()},
      {
        start: currentDate.subtract(1, 'month').firstDayOfMonth(),
        end: currentDate.subtract(1, 'month').lastDayOfMonth()
      },
      {start: currentDate.firstDayOfYear(), end: currentDate.lastDayOfYear()},
    ] as const
    const refOverlayVisible = ref(false)
    watchEffect(() => {
      refOverlayVisible.value = selectedDateRef.value === '自定义日期范围'
    })
    // 表单操作
    const onSubmitCustomDateRange = (e: SubmitEvent) => {
      e.preventDefault()
      refOverlayVisible.value = false
    }
    const onCnacelCustomDateRange = () => {
      refOverlayVisible.value = false
    }
    return () => (
      <MainLayout>
        {{
          title: () => '山竹记账',
          icon: () => <SvgIcon name="menu" class="w-2em h-2em" />,
          default: () => (
            <div class="h-full flex flex-col">
              <SwitchTabs
                selected={selectedDateRef.value}
                v-model:selected={selectedDateRef.value}
                class="[&>nav>menu>li]:(text-start grow-0)">
                <ItemTab tabName="本月" tabId="thisMonth">
                  <ItemSummary
                    startDate={dateList[0].start.format()}
                    endDate={dateList[0].end.format()} />
                </ItemTab>
                <ItemTab tabName="上月" tabId="lastMonth">
                  <ItemSummary
                    startDate={dateList[1].start.format()}
                    endDate={dateList[1].end.format()} />
                </ItemTab>
                <ItemTab tabName="今年" tabId="thisYear">
                  <ItemSummary
                    startDate={dateList[2].start.format()}
                    endDate={dateList[2].end.format()} />
                </ItemTab>
                <ItemTab tabName="自定义日期范围" tabId="customDateRange">
                  <ItemSummary
                    startDate={customDate.start}
                    endDate={customDate.end} />
                </ItemTab>
              </SwitchTabs>

              {/* 自定义日期范围弹出层 */}
              <Overlay
                show={refOverlayVisible.value}
                onClick={() => (refOverlayVisible.value = false)}
                class="flex items-center justify-center b-3 b-solid b-red bg-[var(--overlay-mask-bg)]" >
                <div
                  onClick={(e: MouseEvent) => e.stopPropagation()}
                  class="bg-[var(--overlay-bg)] w-80vw">
                  <header class="bg-[var(--main)] text-[var(--dialog-header-text)] py-12px px-1em">
                    请选择时间
                  </header>
                  <main>
                    <FormContainer onSubmit={onSubmitCustomDateRange}>
                      <FormItem type="date" label="开始时间" v-model={customDate.start} />
                      <FormItem type="date" label="结束时间" v-model={customDate.end} />
                      <footer class="flex justify-between mt-1em children:(w-[calc(50%-.25em)])">
                        <FormItem type="button" onClick={onCnacelCustomDateRange} buttonConfig={{level: 'default', text: '取 消' }}/>
                        <FormItem type="submit-button" buttonConfig={{level: 'primary', text: '确 认' }}/>
                      </footer>
                    </FormContainer>
                  </main>
                </div>
              </Overlay>

              <FloatButton class="fixed bottom-1em right-1em" iconName="add" />
            </div>
          ),
        }}
      </MainLayout>
    )
  },
})
