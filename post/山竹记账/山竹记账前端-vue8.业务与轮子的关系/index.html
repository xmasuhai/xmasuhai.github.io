import { cn } from 'cn'
import { svgIconNames, type SvgIconNames } from '@/components/icons/svgIconNames.ts'

interface SvgIconProps {
name: SvgIconNames
}

/**
* @Description: 图标组件
* @Description: 默认使用插件 vite-plugin-svg-icons-ng
* @Description: 替换使用插件 svgoStore
* @FilePath: content/post/山竹记账/【山竹记账前端-vue】8.业务与轮子的关系.md
*/
export const SvgIcon = defineComponent<SvgIconProps>({
name: 'SvgIcon',
props: {
name: {
  type: String as PropType<SvgIconNames>,
  required: true,
  validator: (value: unknown) =>
    typeof value === 'string' && (svgIconNames as readonly string[]).includes(value),
},
},
setup(props, _ctx) {
// 添加验证
if (!props.name || !(svgIconNames as readonly string[]).includes(props.name)) {
  console.error(`Invalid icon name: ${props.name}`)
}

return () => (
  <svg
    class={cn(
      'where:(h-1em w-1em align-[-0.15em] fill-[currentColor] overflow-clip)',
    )}
    aria-hidden="true">
    <use xlinkHref={`#icon-${props.name}`}/>
  </svg>
)
},
})

interface PlaceCenterProps {

  name: 'PlaceCenter',

      type: String as PropType<PlaceCenterProps['direction']>,
      required: false,
      default: 'horizontal',

  },
  setup( _props, _ctx ) {
    const direction = _props.direction ?? 'horizontal'
    const flexDirection = cn(['horizontal', '-'].includes(direction) ? 'flex-row' : 'flex-col')
    return () => (

        'flex place-items-center place-content-center',
        flexDirection,
      )}>
        {_ctx.slots?.default?.()}
      </section>
    )

})

import { defineStore } from 'pinia'
import { ref } from 'vue'

export const useOverlayStore = defineStore('overlay', () => {
  const overlayVisible = ref(false)

  const toggleOverlay = () => {
    overlayVisible.value = !overlayVisible.value
  }
  const closeOverlay = () => {
    overlayVisible.value = false
  }

  return {
    overlayVisible,
    toggleOverlay,
    closeOverlay,
  }
})
