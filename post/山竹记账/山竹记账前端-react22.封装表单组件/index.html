interface Props {
className?: string
ifUseLabelTag?: boolean
// for 属性，用于内部 input 标签 id 属性联动
labelFor?: string
label?: string
hasLabelColon?: boolean
errorHint?: string
hasErrorPlaceholder?: boolean
placeholder?: string
inputType?: 'text' | 'password' | 'email' | 'number'
value?: string
onChange?: (value: string) => void
onBlur?: () => void
onFocus?: () => void
}

export const InputText: React.FC<Props> = ({
className,
label,
labelFor,
hasLabelColon = true,
ifUseLabelTag,
errorHint,
hasErrorPlaceholder = true,
placeholder,
inputType = 'text',
value,
onChange,
onBlur,
onFocus,
}) => {
const labelForId = labelFor && `form-item-${inputType}-${labelFor}`
const labelChildren = (
<>
  {label}
  {hasLabelColon ? '：' : ''}
</>
)
const labelTag = ifUseLabelTag ? (
<label {...(labelForId ? { htmlFor: labelForId } : {})} className="form-label-sc">
  {labelChildren}
</label>
) : (
<div className="form-label-sc">{labelChildren}</div>
)

return (
<section className={cn('form-item-sc', className)}>
  {label && labelTag}
  <div className="form-item-content-sc">
    <input
      id={labelForId}
      type={inputType}
      value={value}
      onChange={(e) => onChange?.(e.target.value)}
      onBlur={onBlur}
      onFocus={onFocus}
      placeholder={placeholder}
      className={cn(
        'form-input-text-sc',
        errorHint ? 'b-[var(--error-color)]' : 'b-[var(--input-border-color)]'
      )}
    />
  </div>
  {(errorHint || hasErrorPlaceholder) && (
    <span className="form-item-error-sc w-fit buzz">{errorHint ?? <>&nbsp;</>}</span>
  )}
</section>
)
}

  • 清晰地划分表单项结构：外层容器 <section className={cn('form-item-sc', className)}>

    • 标签标题 {label && labelTag}
    • 表单项主体内容 <div className="form-item-content-sc"><input .../></div>
    • 错误提示 <span className="form-item-error-sc w-fit buzz">{errorHint ?? <>&nbsp;</>}</span>

import { EmojiSelector } from '@/components/form/item/emoji/EmojiSelector'
import { cn } from 'cn'
import { useState } from 'react'

interface Props {
  className?: string
  label?: string
  hasLabelColon?: boolean
  errorHint?: string
  hasErrorPlaceholder?: boolean
  value?: string
  onChange?: (value: string) => void
  onBlur?: () => void
  onFocus?: () => void
}

export const InputEmoji: React.FC<Props> = ({
  className,
  label = '符号',
  hasLabelColon,
  errorHint,
  hasErrorPlaceholder,
  value = '',
  onChange,
  onBlur,
  onFocus,
}) => {
  const [emoji, _setEmoji] = useState(value)
  const [emojiHint, setEmojiHint] = useState('')
  const setEmoji = (emoji: string) => {
    _setEmoji(emoji)
    onChange?.(emoji)
  }
  return (
    <section className={cn('form-item-sc relative', className)}>
      <div onBlur={onBlur} onFocus={onFocus} className="form-label-sc text-[22px]">
        <span className="text-[16px]">
          {label} {hasLabelColon ? '：' : ''}
        </span>
        <span className={cn('text-22px inline-block wag')}>{emoji}</span>
        <span className="text-[16px] animate-rainbow color-[var(--vp-c-brand-next)]">
          {emojiHint}
        </span>
      </div>
      <EmojiSelector
        onChangeEmojiUnicode={setEmoji}
        onChangeEmojiHint={setEmojiHint}
        hasError={!!errorHint}
        className="form-item-content-sc"
      />
      {(errorHint || hasErrorPlaceholder) && (
        <span className="form-item-error-sc w-fit buzz">{errorHint ?? <>&nbsp;</>}</span>
      )}
    </section>
  )
}
  • 使用 _setEmoji 拦截器 setEmoji，更新本地组件的状态的同时，使用 onChange 改变父组件数据状态
  • 清晰地划分表单项结构：外层容器 <section className={cn('form-item-sc', className)}>
    • 标签标题包含标题、图标和提示信息
    • 表单项主体内容 <EmojiSelector .../>
    • 错误提示 <span className="form-item-error-sc w-fit buzz">{errorHint ?? <>&nbsp;</>}</span>

注意避免 set-state-in-effect ⇧

  • useEffect 同步代码改变状态，会重渲染。避免导致性能下降的重新渲染现象

import type { JSONValue } from '@/global'
import { create } from 'zustand'
import { formValidate, type FormErrorsType, type FormRules } from '@/lib/validator/formValidate'

export type FormStore<TData, TForm extends JSONValue> = {
  data: TData
  error: FormErrorsType<TForm>
  setData: (data: Partial<TData>) => void
  setError: (error: Partial<FormErrorsType<TForm>>) => void
  clearError: (key?: keyof TForm) => void
  revalidateFormItems: <T extends TForm>(
    formData: Partial<T>,
    formRules: FormRules<T>,
    lastError?: FormErrorsType<T>
  ) => void
}

export function createFormStore<TData, TForm extends JSONValue>(
  options: {
    genInitData: () => TData
    genInitError: () => FormErrorsType<TForm>
  }
) {
  return create<FormStore<TData, TForm>>()((set, get) => ({
    data: options.genInitData(),
    error: options.genInitError(),
    setData: (data: Partial<TData>) => {
      set((state) => ({
        ...state,
        data: { ...state.data, ...data },
      }))
    },
    setError: (error: Partial<FormErrorsType<TForm>>) => {
      set((state) => ({
        ...state,
        error: { ...state.error, ...error },
      }))
    },
    clearError: (key?: keyof TForm) => {
      set((state) => ({
        ...state,
        error: key ? { ...state.error, [key]: [] } : options.genInitError(),
      }))
    },
    revalidateFormItems: <T extends TForm>(
      formData: Partial<T>,
      formRules: FormRules<T>,
      lastError?: FormErrorsType<T>
    ) => {
      get().clearError()
      const keys = Object.keys(formData ?? {})
      const validatingRules = formRules.filter((rule) => keys.includes(String(rule.key)))
      const formError = formValidate(formData, validatingRules)
      get().setError({ ...lastError, ...formError } as Partial<FormErrorsType<TForm>>)
    },
  }))
}
